    60→def add_styled_table(doc, headers, rows, col_widths=None):
    61→    """Add a professionally styled table."""
    62→    table = doc.add_table(rows=1 + len(rows), cols=len(headers))
    63→    table.alignment = WD_TABLE_ALIGNMENT.CENTER
    64→    table.style = 'Table Grid'
    65→
    66→    # Header row
    67→    for i, header in enumerate(headers):
    68→        cell = table.rows[0].cells[i]
    69→        cell.text = ""
    70→        p = cell.paragraphs[0]
    71→        run = p.add_run(header)
    72→        run.bold = True
    73→        run.font.size = Pt(10)
    74→        run.font.color.rgb = c(P["white"])
    75→        run.font.name = "Calibri"
    76→        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    77→        set_cell_shading(cell, P["primary"])
    78→
    79→    # Data rows
    80→    for row_idx, row_data in enumerate(rows):
    81→        for col_idx, cell_text in enumerate(row_data):
    82→            cell = table.rows[row_idx + 1].cells[col_idx]
    83→            cell.text = ""
    84→            p = cell.paragraphs[0]
    85→            run = p.add_run(str(cell_text))
    86→            run.font.size = Pt(9.5)
    87→            run.font.name = "Calibri"
    88→            run.font.color.rgb = c(P["body"])
    89→            if row_idx % 2 == 1:
    90→                set_cell_shading(cell, P["surface"])
    91→
    92→    # Set column widths if provided
    93→    if col_widths:
    94→        for row in table.rows:
    95→            for idx, width in enumerate(col_widths):
    96→                row.cells[idx].width = Cm(width)
    97→
    98→    return table
