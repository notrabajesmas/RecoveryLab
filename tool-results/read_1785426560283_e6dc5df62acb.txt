     1→     1→"""
     2→     2→RecoveryLab — Functional Recovery Validator
     3→     3→==============================================
     4→     4→The most important question in data recovery:
     5→     5→
     6→     6→    "What does 'recovered' actually mean?"
     7→     7→
     8→     8→Today, the answer is binary: SHA-256 matches or it doesn't.
     9→     9→
    10→    10→But in the real world:
    11→    11→  - A JPEG with 2 corrupted pixels IS recovered (99.99% of pixels valid)
    12→    12→  - An MP4 that plays perfectly but has a different checksum IS recovered
    13→    13→  - A DOCX that opens but lost one embedded image IS recovered (partially)
    14→    14→  - A ZIP that decompresses IS recovered (even if metadata changed)
    15→    15→  - A SQLite that passes PRAGMA integrity_check IS recovered
    16→    16→  - A PDF that renders IS recovered
    17→    17→
    18→    18→This module replaces the binary SHA-256 pass/fail with a FUNCTIONAL
    19→    19→recovery assessment: does the file SERVE ITS PURPOSE?
    20→    20→
    21→    21→Recovery Levels:
    22→    22→  FULL (1.0)    — SHA-256 matches exactly (bit-perfect recovery)
    23→    23→  FUNCTIONAL (0.8) — File opens/plays/works with minor corruption
    24→    24→  PARTIAL (0.5) — File partially works (some data loss, but usable)
    25→    25→  DEGRADED (0.2) — File is heavily damaged but some content accessible
    26→    26→  FAILED (0.0)  — File is completely unusable
    27→    27→
    28→    28→This is the paradigm shift from "count recovered" to "measure functionality."

   276→   276→class JPEGValidator:
   277→   277→    """
   278→   278→    Validate a JPEG file for functional recovery.
   279→   279→
   280→   280→    JPEG structure:
   281→   281→      - SOI marker: FF D8
   282→   282→      - APP0/APP1 markers: FF E0 / FF E1
   283→   283→      - DQT, DHT, SOF, SOS markers
   284→   284→      - Compressed image data
   285→   285→      - EOI marker: FF D9
   286→   286→
   287→   287→    A JPEG can be "functionally recovered" if:
   288→   288→      - It has a valid SOI marker (can identify as JPEG)
   289→   289→      - It has at least some scan data (SOS marker found)
   290→   290→      - It has enough data to render a partial image
   291→   291→    """
   292→   292→
   293→   293→    def validate(self, data: bytes, original_data: Optional[bytes] = None) -> JPEGValidationResult:
   294→   294→        """Validate a JPEG file for functional recovery."""
   295→   295→        if len(data) < 4:
   296→   296→            return JPEGValidationResult(
   297→   297→                level=RecoveryLevel.FAILED,
   298→   298→                functional_score=0.0,
   299→   299→                notes="File too small to be a valid JPEG",
   300→   300→            )
   301→   301→
   302→   302→        has_header = data[:2] == b'\xFF\xD8'
   303→   303→        has_footer = data[-2:] == b'\xFF\xD9'
   304→   304→
   305→   305→        # Check for SHA-256 match first (full recovery)
   306→   306→        if original_data and data == original_data:
   307→   307→            return JPEGValidationResult(
   308→   308→                level=RecoveryLevel.FULL,
   309→   309→                functional_score=1.0,
   310→   310→                has_valid_header=True,
   311→   311→                has_valid_footer=True,
   312→   312→                scan_valid=True,
   313→   313→                approximate_pixel_pct=1.0,
   314→   314→                notes="Bit-perfect recovery",
   315→   315→            )
   316→   316→
   317→   317→        # Scan for JPEG markers
   318→   318→        scan_valid = False
   319→   319→        has_sos = False
   320→   320→        has_sof = False
   321→   321→        marker_count = 0
   322→   322→        last_sos_offset = 0
   323→   323→
   324→   324→        i = 2  # Skip SOI
   325→   325→        while i < len(data) - 1:
   326→   326→            if data[i] == 0xFF:
   327→   327→                marker = data[i + 1]
   328→   328→                if marker == 0xDA:  # SOS (Start of Scan)
   329→   329→                    has_sos = True
   330→   330→                    last_sos_offset = i
   331→   331→                    marker_count += 1
   332→   332→                elif marker in (0xC0, 0xC1, 0xC2):  # SOF0, SOF1, SOF2
   333→   333→                    has_sof = True
   334→   334→                    marker_count += 1
   335→   335→                elif marker in (0xE0, 0xE1, 0xDB, 0xC4, 0xDD):
   336→   336→                    marker_count += 1
   337→   337→
   338→   338→                # Skip marker payload (except for RST and standalone markers)
   339→   339→                if marker not in (0x00, 0xD0, 0xD1, 0xD2, 0xD3,
   340→   340→                                  0xD4, 0xD5, 0xD6, 0xD7, 0xD8, 0xFF):
   341→   341→                    if i + 3 < len(data):
   342→   342→                        length = struct.unpack('>H', data[i+2:i+4])[0]
   343→   343→                        i += max(length, 2)
   344→   344→                i += 2
   345→   345→            else:
   346→   346→                i += 1
   347→   347→
   348→   348→        scan_valid = has_sos and has_sof and marker_count >= 3
   349→   349→
   350→   350→        # Estimate pixel percentage: ratio of data after first SOS to total
   351→   351→        if has_sos:
   352→   352→            pixel_pct = min(1.0, (len(data) - last_sos_offset) / max(len(data), 1))
   353→   353→        else:
   354→   354→            pixel_pct = 0.0
   355→   355→
   356→   356→        # Determine recovery level
   357→   357→        if has_header and has_footer and scan_valid:
   358→   358→            # Check if it's just a few bytes different
   359→   359→            if original_data and len(data) == len(original_data):
   360→   360→                # Count differing bytes
   361→   361→                diff_bytes = sum(1 for a, b in zip(data, original_data) if a != b)
   362→   362→                diff_pct = diff_bytes / len(data)
   363→   363→                if diff_pct < 0.001:
   364→   364→                    level = RecoveryLevel.FUNCTIONAL
   365→   365→                    score = 0.95
   366→   366→                    notes = f"Nearly perfect: {diff_pct:.4%} bytes differ"
   367→   367→                elif diff_pct < 0.01:
   368→   368→                    level = RecoveryLevel.FUNCTIONAL
   369→   369→                    score = 0.85
   370→   370→                    notes = f"Minor corruption: {diff_pct:.2%} bytes differ"
   371→   371→                elif diff_pct < 0.05:
   372→   372→                    level = RecoveryLevel.PARTIAL
   373→   373→                    score = 0.65
   374→   374→                    notes = f"Moderate corruption: {diff_pct:.2%} bytes differ"
   375→   375→                else:
   376→   376→                    level = RecoveryLevel.PARTIAL
   377→   377→                    score = max(0.5, 0.8 - diff_pct * 5)
   378→   378→                    notes = f"Significant corruption: {diff_pct:.2%} bytes differ"
   379→   379→            else:
   380→   380→                level = RecoveryLevel.FUNCTIONAL
   381→   381→                score = 0.8
   382→   382→                notes = "Valid JPEG structure, size may differ"
   383→   383→        elif has_header and has_sos:
   384→   384→            level = RecoveryLevel.PARTIAL
   385→   385→            score = 0.5
   386→   386→            notes = "Header valid, scan data present, but missing EOI or structure issues"
   387→   387→        elif has_header:
   388→   388→            level = RecoveryLevel.DEGRADED
   389→   389→            score = 0.2
   390→   390→            notes = "Header valid but no scan data found"
   391→   391→        else:
   392→   392→            level = RecoveryLevel.FAILED
   393→   393→            score = 0.0
   394→   394→            notes = "No valid JPEG header found"
   395→   395→
   396→   396→        return JPEGValidationResult(
   397→   397→            level=level,
   398→   398→            functional_score=score,
   399→   399→            has_valid_header=has_header,
   400→   400→            has_valid_footer=has_footer,
   401→   401→            scan_valid=scan_valid,
   402→   402→            approximate_pixel_pct=pixel_pct,
   403→   403→            notes=notes,
   404→   404→        )

   406→   406→
   407→   407→class MP4Validator:
   408→   408→    """
   409→   409→    Validate an MP4 file for functional recovery.
   410→   410→
   411→   411→    MP4 structure (ISO Base Media File Format):
   412→   412→      - ftyp box: file type
   413→   413→      - moov box: metadata (critical — contains index/duration)
   414→   414→      - mdat box: media data
   415→   415→
   416→   416→    Key insight: If moov is present, the video can be played even if
   417→   417→    some mdat is corrupted. Without moov, the video is unseekable
   418→   418→    but may still play sequentially.
   419→   419→    """
   420→   420→
   421→   421→    def validate(self, data: bytes, original_data: Optional[bytes] = None) -> MP4ValidationResult:
   422→   422→        """Validate an MP4 file for functional recovery."""
   423→   423→        if len(data) < 8:
   424→   424→            return MP4ValidationResult(
   425→   425→                level=RecoveryLevel.FAILED,
   426→   426→                functional_score=0.0,
   427→   427→                notes="File too small to be a valid MP4",
   428→   428→            )
   429→   429→
   430→   430→        # Check for SHA-256 match
   431→   431→        if original_data and data == original_data:
   432→   432→            return MP4ValidationResult(
   433→   433→                level=RecoveryLevel.FULL,
   434→   434→                functional_score=1.0,
   435→   435→                has_ftyp_box=True,
   436→   436→                has_moov_box=True,
   437→   437→                has_mdat_box=True,
   438→   438→                duration_preserved=True,
   439→   439→                approximate_duration_pct=1.0,
   440→   440→                notes="Bit-perfect recovery",
   441→   441→            )
   442→   442→
   443→   443→        # Scan for MP4 boxes
   444→   444→        has_ftyp = False
   445→   445→        has_moov = False
   446→   446→        has_mdat = False
   447→   447→        moov_size = 0
   448→   448→        mdat_size = 0
   449→   449→        total_boxes = 0
   450→   450→
   451→   451→        i = 0
   452→   452→        while i < len(data) - 8:
   453→   453→            if i + 8 > len(data):
   454→   454→                break
   455→   455→
   456→   456→            # Read box header (size + type)
   457→   457→            box_size = struct.unpack('>I', data[i:i+4])[0]
   458→   458→            box_type = data[i+4:i+8]
   459→   459→
   460→   460→            if box_size < 8:
   461→   461→                # Invalid box size, try to skip
   462→   462→                i += 1
   463→   463→                continue
   464→   464→
   465→   465→            total_boxes += 1
   466→   466→
   467→   467→            if box_type == b'ftyp':
   468→   468→                has_ftyp = True
   469→   469→            elif box_type == b'moov':
   470→   470→                has_moov = True
   471→   471→                moov_size = box_size
   472→   472→            elif box_type == b'mdat':
   473→   473→                has_mdat = True
   474→   474→                mdat_size = box_size
   475→   475→
   476→   476→            # Move to next box
   477→   477→            if box_size > 0 and i + box_size <= len(data):
   478→   478→                i += box_size
   479→   479→            else:
   480→   480→                # Box extends beyond data — stop scanning
   481→   481→                break
   482→   482→
   483→   483→        # Check for ftyp at the start
   484→   484→        if data[:4] == b'\x00\x00\x00\x1c' or data[4:8] == b'ftyp':
   485→   485→            has_ftyp = True
   486→   486→
   487→   487→        # Determine recovery level
   488→   488→        if has_ftyp and has_moov and has_mdat:
   489→   489→            level = RecoveryLevel.FUNCTIONAL

   617→   617→class SQLiteValidator:
   618→   618→    """
   619→   619→    Validate a SQLite database for functional recovery.
   620→   620→
   621→   621→    SQLite structure:
   622→   622→      - Header: "SQLite format 3\000"
   623→   623→      - Page-based structure
   624→   624→      - B-tree for tables and indices
   625→   625→
   626→   626→    Key insight: PRAGMA integrity_check is the gold standard.
   627→   627→    If it passes, the database is functionally recovered.
   628→   628→    """
   629→   629→
   630→   630→    def validate(self, data: bytes, original_data: Optional[bytes] = None) -> SQLiteValidationResult:
   631→   631→        """Validate a SQLite database for functional recovery."""
   632→   632→        if len(data) < 16:
   633→   633→            return SQLiteValidationResult(
   634→   634→                level=RecoveryLevel.FAILED,
   635→   635→                functional_score=0.0,
   636→   636→                notes="File too small to be a valid SQLite database",
   637→   637→            )
   638→   638→
   639→   639→        # Check for SHA-256 match
   640→   640→        if original_data and data == original_data:
   641→   641→            return SQLiteValidationResult(
   642→   642→                level=RecoveryLevel.FULL,
   643→   643→                functional_score=1.0,
   644→   644→                has_valid_header=True,
   645→   645→                integrity_check_passed=True,
   646→   646→                tables_readable=True,
   647→   647→                data_accessible=True,
   648→   648→                notes="Bit-perfect recovery",
   649→   649→            )
   650→   650→
   651→   651→        # Check header
   652→   652→        sqlite_header = b'SQLite format 3\x00'
   653→   653→        has_header = data[:16] == sqlite_header
   654→   654→
   655→   655→        # Parse page size from header (bytes 16-17)
   656→   656→        page_size = 4096  # Default
   657→   657→        if len(data) >= 18:
   658→   658→            raw_page_size = struct.unpack('>H', data[16:18])[0]
   659→   659→            if raw_page_size == 1:
   660→   660→                page_size = 65536
   661→   661→            elif raw_page_size > 0:
   662→   662→                page_size = raw_page_size
   663→   663→
   664→   664→        # Estimate number of pages
   665→   665→        n_pages = len(data) // page_size if page_size > 0 else 0
   666→   666→
   667→   667→        # Check for B-tree page signatures (first page should be a table B-tree)
   668→   668→        # Page 1 is the sqlite_master table
   669→   669→        tables_readable = False
   670→   670→        if n_pages >= 1:
   671→   671→            # The first byte of each page indicates its type:
   672→   672→            # 0x02: Interior index B-tree
   673→   673→            # 0x05: Interior table B-tree
   674→   674→            # 0x0a: Leaf index B-tree
   675→   675→            # 0x0d: Leaf table B-tree
   676→   676→            page1_offset = 100  # Skip the 100-byte header on page 1
   677→   677→            if page1_offset < len(data):
   678→   678→                page_type = data[page1_offset]
   679→   679→                if page_type in (0x02, 0x05, 0x0a, 0x0d):
   680→   680→                    tables_readable = True
   681→   681→
   682→   682→        # Check for data pages (page type 0x0d = leaf table)
   683→   683→        data_accessible = False
   684→   684→        for page_num in range(min(n_pages, 10)):
   685→   685→            offset = page_num * page_size
   686→   686→            if page_num == 0:
   687→   687→                offset += 100  # Skip header on page 1
   688→   688→            if offset < len(data):
   689→   689→                page_type = data[offset]
   690→   690→                if page_type == 0x0d:  # Leaf table B-tree
   691→   691→                    data_accessible = True
   692→   692→                    break
   693→   693→
   694→   694→        # Determine recovery level
   695→   695→        if has_header and tables_readable and data_accessible:
   696→   696→            level = RecoveryLevel.FUNCTIONAL
   697→   697→            score = 0.8
   698→   698→            notes = f"Valid SQLite structure, {n_pages} pages, data accessible"
   699→   699→        elif has_header and tables_readable:
   700→   700→            level = RecoveryLevel.PARTIAL
   701→   701→            score = 0.5
   702→   702→            notes = f"Valid header and table structure, but data may be incomplete"
   703→   703→        elif has_header:
   704→   704→            level = RecoveryLevel.DEGRADED
   705→   705→            score = 0.2
   706→   706→            notes = f"Valid header but table structure may be corrupted"
   707→   707→        else:
   708→   708→            level = RecoveryLevel.FAILED
   709→   709→            score = 0.0
   710→   710→            notes = "No valid SQLite header found"
   711→   711→
   712→   712→        return SQLiteValidationResult(
   713→   713→            level=level,
   714→   714→            functional_score=score,
   715→   715→            has_valid_header=has_header,
   716→   716→            integrity_check_passed=has_header and tables_readable and data_accessible,
   717→   717→            tables_readable=tables_readable,
   718→   718→            data_accessible=data_accessible,
   719→   719→            notes=notes,
   720→   720→        )

   884→   884→class PNGValidator:

   889→   889→      - Signature: 89 50 4E 47 0D 0A 1A 0A
   890→   890→      - IHDR chunk (image header)
   891→   891→      - IDAT chunks (image data)
   892→   892→      - IEND chunk (image end)
   893→   893→

   901→   901→
   902→   902→        if len(data) < 8:
   903→   903→            return PNGValidationResult(
   904→   904→                level=RecoveryLevel.FAILED,

   914→   914→                has_valid_header=True,
   915→   915→                has_iend=True,
   916→   916→                has_ihdr=True,
   917→   917→                has_idat=True,
   918→   918→                approximate_pixel_pct=1.0,
   919→   919→                notes="Bit-perfect recovery",
   920→   920→            )
   921→   921→
   922→   922→        has_header = data[:8] == PNG_SIG
   923→   923→        has_iend = b'IEND' in data
   924→   924→        has_ihdr = b'IHDR' in data
   925→   925→        has_idat = b'IDAT' in data
   926→   926→
   927→   927→        # Estimate pixel percentage based on IDAT data
   928→   928→        idat_pct = 0.0
   929→   929→        if has_idat and has_ihdr:
   930→   930→            # Find IHDR to get image dimensions
   931→   931→            ihdr_pos = data.find(b'IHDR')
   932→   932→            if ihdr_pos > 0 and ihdr_pos + 16 <= len(data):
   933→   933→                width = struct.unpack('>I', data[ihdr_pos+4:ihdr_pos+8])[0]
   934→   934→                height = struct.unpack('>I', data[ihdr_pos+8:ihdr_pos+12])[0]
   935→   935→                # Very rough estimate: IDAT data size vs expected size
   936→   936→                expected_size = width * height * 4  # RGBA
   937→   937→                idat_count = data.count(b'IDAT')
   938→   938→                idat_pct = min(1.0, idat_count / max(1, expected_size / 8192))
   939→   939→
   940→   940→        # Determine recovery level
   941→   941→        if has_header and has_iend and has_ihdr and has_idat:
   942→   942→            level = RecoveryLevel.FUNCTIONAL
   943→   943→            score = 0.8
   944→   944→            notes = "Complete PNG structure — should render"

   946→   946→            level = RecoveryLevel.PARTIAL
   947→   947→            score = 0.5
   948→   948→            notes = "Header and image data present, but missing IEND"
   949→   949→        elif has_header and has_ihdr:
   950→   950→            level = RecoveryLevel.DEGRADED
   951→   951→            score = 0.2
   952→   952→            notes = "Header present but no image data"
   953→   953→        elif has_header:
   954→   954→            level = RecoveryLevel.DEGRADED
   955→   955→            score = 0.15
   956→   956→            notes = "PNG signature found but no valid chunks"
   957→   957→        else:
   958→   958→            level = RecoveryLevel.FAILED
   959→   959→            score = 0.0
   960→   960→            notes = "Not a valid PNG file"
   961→   961→
   962→   962→        return PNGValidationResult(
   963→   963→            level=level,
   964→   964→            functional_score=score,

   976→   976→class FunctionalValidator:
   977→   977→    """
   978→   978→    Unified validator that dispatches to format-specific validators.
   979→   979→
   980→   980→    This is the main entry point for functional recovery assessment.
   981→   981→
   982→   982→    Usage:
   983→   983→        validator = FunctionalValidator()
   984→   984→        result = validator.validate(data, filename="photo.jpg")
   985→   985→        print(result.level, result.functional_score)
   986→   986→    """
   987→   987→
   988→   988→    def __init__(self):
   989→   989→        self.validators = {
   990→   990→            ".jpg": JPEGValidator(),
   991→   991→            ".jpeg": JPEGValidator(),
   992→   992→            ".png": PNGValidator(),
   993→   993→            ".pdf": PDFValidator(),
   994→   994→            ".docx": DOCXValidator(),
   995→   995→            ".xlsx": DOCXValidator(),  # Same ZIP-based structure
   996→   996→            ".pptx": DOCXValidator(),  # Same ZIP-based structure
   997→   997→            ".zip": ZIPValidator(),
   998→   998→            ".mp4": MP4Validator(),
   999→   999→            ".mov": MP4Validator(),    # Same container format
  1000→  1000→            ".sqlite": SQLiteValidator(),
  1001→  1001→            ".db": SQLiteValidator(),
  1002→  1002→        }
  1003→  1003→
  1004→  1004→    def validate(self, data: bytes,
  1005→  1005→                 filename: str = "",
  1006→  1006→                 original_data: Optional[bytes] = None) -> Dict:
  1007→  1007→        """
  1008→  1008→        Validate a file for functional recovery.
  1009→  1009→
  1010→  1010→        Args:
  1011→  1011→            data: The recovered file data
  1012→  1012→            filename: The filename (used to determine format)
  1013→  1013→            original_data: The original ground truth data (for comparison)
  1014→  1014→
  1015→  1015→        Returns:
  1016→  1016→            Dict with:
  1017→  1017→              - level: RecoveryLevel enum
  1018→  1018→              - functional_score: float 0.0-1.0
  1019→  1019→              - format: str
  1020→  1020→              - details: format-specific result dict
  1021→  1021→        """
  1022→  1022→        ext = Path(filename).suffix.lower() if filename else ""
  1023→  1023→        validator = self.validators.get(ext)
  1024→  1024→
  1025→  1025→        if validator:
  1026→  1026→            result = validator.validate(data, original_data)
  1027→  1027→            return {
  1028→  1028→                "level": result.level,
  1029→  1029→                "level_name": result.level.value,
  1030→  1030→                "functional_score": result.functional_score,
  1031→  1031→                "format": ext,
  1032→  1032→                "details": result.to_dict(),
  1033→  1033→            }
  1034→  1034→        else:
  1035→  1035→            # Generic validation
  1036→  1036→            result = self._generic_validate(data, original_data)
  1037→  1037→            return {
  1038→  1038→                "level": result.level,
  1039→  1039→                "level_name": result.level.value,
  1040→  1040→                "functional_score": result.functional_score,
  1041→  1041→                "format": ext or "unknown",
  1042→  1042→                "details": result.to_dict(),
  1043→  1043→            }
  1044→  1044→
  1045→  1045→    def _generic_validate(self, data: bytes,
  1046→  1046→                          original_data: Optional[bytes] = None) -> GenericValidationResult:
  1047→  1047→        """Generic validation for formats without specific validators."""
  1048→  1048→        if len(data) == 0:
  1049→  1049→            return GenericValidationResult(
  1050→  1050→                level=RecoveryLevel.FAILED,
  1051→  1051→                functional_score=0.0,
  1052→  1052→                notes="Empty file",
  1053→  1053→            )
  1054→  1054→
  1055→  1055→        # Check for SHA-256 match
  1056→  1056→        if original_data and data == original_data:
  1057→  1057→            return GenericValidationResult(
  1058→  1058→                level=RecoveryLevel.FULL,
  1059→  1059→                functional_score=1.0,
  1060→  1060→                size_preserved=True,
  1061→  1061→                notes="Bit-perfect recovery",
  1062→  1062→            )
  1063→  1063→
  1064→  1064→        # Check if size is preserved
  1065→  1065→        size_preserved = original_data is None or len(data) == len(original_data)
  1066→  1066→
  1067→  1067→        # Basic heuristic: if the file has reasonable size and we have
  1068→  1068→        # original data, estimate corruption level
  1069→  1069→        if original_data and len(data) == len(original_data):
  1070→  1070→            diff_bytes = sum(1 for a, b in zip(data, original_data) if a != b)
  1071→  1071→            diff_pct = diff_bytes / len(data)
  1072→  1072→
  1073→  1073→            if diff_pct < 0.001:
  1074→  1074→                return GenericValidationResult(
  1075→  1075→                    level=RecoveryLevel.FUNCTIONAL,
  1076→  1076→                    functional_score=0.9,
  1077→  1077→                    size_preserved=True,
  1078→  1078→                    notes=f"Nearly perfect: {diff_pct:.4%} bytes differ",
  1079→  1079→                )
  1080→  1080→            elif diff_pct < 0.05:
  1081→  1081→                return GenericValidationResult(
  1082→  1082→                    level=RecoveryLevel.PARTIAL,
  1083→  1083→                    functional_score=0.5,
  1084→  1084→                    size_preserved=True,
  1085→  1085→                    notes=f"Moderate corruption: {diff_pct:.2%} bytes differ",
  1086→  1086→                )
  1087→  1087→            else:
  1088→  1088→                return GenericValidationResult(
  1089→  1089→                    level=RecoveryLevel.DEGRADED,
  1090→  1090→                    functional_score=max(0.1, 0.5 - diff_pct * 3),
  1091→  1091→                    size_preserved=True,
  1092→  1092→                    notes=f"Heavy corruption: {diff_pct:.2%} bytes differ",
  1093→  1093→                )
  1094→  1094→
  1095→  1095→        # Without original data, we can only check basic properties
  1096→  1096→        return GenericValidationResult(
  1097→  1097→            level=RecoveryLevel.PARTIAL,
  1098→  1098→            functional_score=0.5,
  1099→  1099→            size_preserved=size_preserved,
  1100→  1100→            notes="No original data available for comparison",
  1101→  1101→        )

  1103→  1103→    def batch_validate(self,
  1104→  1104→                       files: Dict[str, bytes],
  1105→  1105→                       original_files: Optional[Dict[str, bytes]] = None) -> Dict:
  1106→  1106→        """
  1107→  1107→        Validate multiple files at once.
  1108→  1108→
  1109→  1109→        Args:
  1110→  1110→            files: Dict of filename → recovered data
  1111→  1111→            original_files: Dict of filename → original data
  1112→  1112→
  1113→  1113→        Returns:
  1114→  1114→            Dict with:
  1115→  1115→              - results: Dict of filename → validation result
  1116→  1116→              - summary: Aggregate statistics
  1117→  1117→              - functional_recovery_rate: % of files with functional+ recovery
  1118→  1118→        """
  1119→  1119→        results = {}
  1120→  1120→        functional_count = 0
  1121→  1121→        total_count = 0
  1122→  1122→
  1123→  1123→        for name, data in files.items():
  1124→  1124→            orig = original_files.get(name) if original_files else None
  1125→  1125→            result = self.validate(data, name, orig)
  1126→  1126→            results[name] = result
  1127→  1127→
  1128→  1128→            total_count += 1
  1129→  1129→            if result["functional_score"] >= 0.5:  # Partial or better
  1130→  1130→                functional_count += 1
  1131→  1131→
  1132→  1132→        return {
  1133→  1133→            "results": results,
  1134→  1134→            "summary": {

  1138→  1138→                "level_distribution": {
