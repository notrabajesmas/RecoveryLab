   142→
   143→@dataclass
   144→class CrossoverResult:
   145→    """Complete crossover curve result."""
   146→    dataset: str
   147→    timestamp: str
   148→    points: List[CrossoverPoint] = field(default_factory=list)
   149→    crossover_point: Optional[float] = None
   150→    crossover_type: str = ""  # "gradual" or "abrupt"
   151→
   152→    def find_crossover(self) -> Optional[float]:
   153→        """Find the MFT damage % where Carving overtakes MFT-First."""
   154→        for i in range(1, len(self.points)):
   155→            prev = self.points[i - 1]
   156→            curr = self.points[i]
   157→
   158→            # Crossover: MFT was winning, now Carving is winning (or equal)
   159→            if prev.delta_recovery > 0 and curr.delta_recovery <= 0:
   160→                # Linear interpolation
   161→                if prev.delta_recovery - curr.delta_recovery != 0:
   162→                    frac = prev.delta_recovery / (prev.delta_recovery - curr.delta_recovery)
   163→                    self.crossover_point = prev.mft_damage_pct + frac * (curr.mft_damage_pct - prev.mft_damage_pct)
   164→                else:
   165→                    self.crossover_point = curr.mft_damage_pct
   166→
   167→                # Determine if abrupt or gradual
   168→                if i > 0 and abs(prev.delta_recovery - curr.delta_recovery) > 0.3:
   169→                    self.crossover_type = "abrupt"
   170→                else:
   171→                    self.crossover_type = "gradual"
   172→
   173→                return self.crossover_point
   174→
   175→        return None
   176→
   177→    def to_dict(self) -> Dict:
   178→        return {
   179→            "dataset": self.dataset,
   180→            "timestamp": self.timestamp,
   181→            "crossover_point": round(self.crossover_point, 4) if self.crossover_point else None,
   182→            "crossover_type": self.crossover_type,
   183→            "points": [p.to_dict() for p in self.points],
   184→        }

   187→def compute_statistics(carving_recoveries: List[float],
   188→                        mft_recoveries: List[float]) -> Dict:
   189→    """
   190→    Compute statistical analysis for a single crossover point.
   191→
   192→    Returns:
   193→      - 95% CI for each strategy
   194→      - p-value (two-tailed t-test)
   195→      - Effect size (Cohen's d)
   196→    """
   197→    import math
   198→
   199→    n = len(carving_recoveries)
   200→    if n < 2:
   201→        return {
   202→            "ci_lower_carving": 0.0, "ci_upper_carving": 0.0,
   203→            "ci_lower_mft": 0.0, "ci_upper_mft": 0.0,
   204→            "p_value": 1.0, "effect_size": 0.0,
   205→        }
   206→
   207→    # Mean and std for each
   208→    mean_c = sum(carving_recoveries) / n
   209→    mean_m = sum(mft_recoveries) / n
   210→
   211→    var_c = sum((x - mean_c) ** 2 for x in carving_recoveries) / (n - 1) if n > 1 else 0
   212→    var_m = sum((x - mean_m) ** 2 for x in mft_recoveries) / (n - 1) if n > 1 else 0
   213→
   214→    std_c = math.sqrt(var_c)
   215→    std_m = math.sqrt(var_m)
   216→
   217→    # 95% CI (using t-distribution approximation for small n)
   218→    # For n >= 30, use 1.96; for small n, use t-value
   219→    t_values = {2: 12.706, 3: 4.303, 4: 3.182, 5: 2.776, 10: 2.262,
   220→                20: 2.093, 30: 2.042, 50: 2.009, 100: 1.984}
   221→    if n in t_values:
   222→        t = t_values[n]
   223→    elif n < 2:
   224→        t = 12.706
   225→    elif n < 30:
   226→        t = 2.776  # Conservative
   227→    else:
   228→        t = 1.96
   229→
   230→    se_c = std_c / math.sqrt(n) if n > 0 else 0
   231→    se_m = std_m / math.sqrt(n) if n > 0 else 0
   232→
   233→    # Paired t-test
   234→    diffs = [c - m for c, m in zip(carving_recoveries, mft_recoveries)]
   235→    mean_diff = sum(diffs) / n
   236→    var_diff = sum((d - mean_diff) ** 2 for d in diffs) / (n - 1) if n > 1 else 0
   237→    se_diff = math.sqrt(var_diff / n) if n > 0 else 0
   238→
   239→    if se_diff > 0:
   240→        t_stat = mean_diff / se_diff
   241→        # Approximate p-value (two-tailed)
   242→        # For large |t|, p is very small
   243→        p_value = 2.0 * (1.0 - _t_cdf_approx(abs(t_stat), n - 1))
   244→        p_value = max(0.0, min(1.0, p_value))
   245→    else:
   246→        p_value = 1.0 if abs(mean_diff) < 0.001 else 0.0
   247→
   248→    # Cohen's d
   249→    pooled_std = math.sqrt((var_c + var_m) / 2) if (var_c + var_m) > 0 else 1.0
   250→    cohens_d = (mean_m - mean_c) / pooled_std if pooled_std > 0 else 0.0
   251→
   252→    return {
   253→        "ci_lower_carving": mean_c - t * se_c,
   254→        "ci_upper_carving": mean_c + t * se_c,
   255→        "ci_lower_mft": mean_m - t * se_m,
   256→        "ci_upper_mft": mean_m + t * se_m,
   257→        "p_value": p_value,
   258→        "effect_size": cohens_d,
   259→    }
   260→
   261→
   262→def _t_cdf_approx(t: float, df: int) -> float:
   263→    """Approximate CDF of t-distribution using normal approximation."""
   264→    import math
   265→    if df >= 30:
   266→        # Normal approximation
   267→        return 0.5 * (1 + math.erf(t / math.sqrt(2)))
   268→    else:
   269→        # Simple approximation
   270→        x = (t + 0.044715 * t**3) / math.sqrt(1 + t**2 / df)
   271→        return 0.5 * (1 + math.erf(x / math.sqrt(2)))

   274→def run_crossover_curve(
   275→    image: bytes,
   276→    manifest: Dict,
   277→    seed: int = 42,
   278→    step: float = 0.05,
   279→    max_damage: float = 1.0,
   280→    n_repetitions: int = 1,
   281→    read_budget: int = 0,
   282→) -> CrossoverResult:
   283→    """
   284→    Run the crossover curve experiment.
   285→
   286→    For each damage level (0%, 5%, 10%, ..., 100%):
   287→      1. Apply MFT partial deletion
   288→      2. Run Carving, MFT-First, Motor C
   289→      3. Judge results
   290→      4. Record data point
   291→      5. If n_repetitions > 1, repeat and compute statistics
   292→
   293→    Returns a CrossoverResult with all data points.
   294→    """
   295→    dataset_id = manifest.get("serial", "unknown")
   296→    result = CrossoverResult(
   297→        dataset=dataset_id,
   298→        timestamp=datetime.now(timezone.utc).isoformat(),
   299→    )
   300→
   301→    motor_carving = MotorCarving()
   302→    motor_mft = MotorBMFTFirst()
   303→    motor_c = MotorCOrchestrator()
   304→    judge = RecoveryJudge(manifest)
   305→
   306→    # Generate damage levels
   307→    damage_levels = []
   308→    d = 0.0
   309→    while d <= max_damage + 0.001:
   310→        damage_levels.append(round(d, 2))
   311→        d += step
   312→
   313→    print(f"\n{'='*90}")
   314→    print(f"STRATEGY CROSSOVER CURVE — {dataset_id}")
   315→    print(f"{'='*90}")
   316→    print(f"  Damage levels: {len(damage_levels)}")
   317→    print(f"  Step: {step:.0%}")
   318→    print(f"  Repetitions per point: {n_repetitions}")
   319→    print(f"{'─'*90}")
   320→    print(f"  {'Damage':>8s} │ {'Conf':>6s} │ {'Carving':>10s} │ {'MFT-1st':>10s} │ "
   321→          f"{'Motor C':>10s} │ {'Δ Rec':>8s} │ {'Winner':>10s} │ {'p-val':>8s} │ {'d':>6s}")
   322→    print(f"{'─'*90}")
   323→
   324→    for damage_pct in damage_levels:
   325→        carving_recoveries = []
   326→        mft_recoveries = []
   327→        motor_c_recoveries = []
   328→
   329→        # Aggregated metrics (averaged over repetitions)
   330→        sum_carving = None
   331→        sum_mft = None
   332→        sum_motor_c = None
   333→
   334→        for rep in range(n_repetitions):
   335→            # Apply corruption
   336→            rep_seed = seed + rep * 1000  # Different seed per repetition
   337→            if damage_pct == 0.0:
   338→                corrupted_image = image
   339→                corruption_meta = {}
   340→            else:
   341→                model = MFTPartialDeleteModel(seed=rep_seed)
   342→                image_copy = bytearray(image)
   343→                corr_result = model.apply(image_copy, manifest, severity=damage_pct)
   344→                corrupted_image = corr_result.corrupted_image
   345→                corruption_meta = corr_result.manifest_corruption
   346→
   347→            # Calculate MFT confidence
   348→            confidence = motor_c.compute_mft_confidence(corrupted_image, manifest)
   349→
   350→            # ─── Carving ──────────────────────────────────────────────
   351→            result_carving = motor_carving.recover(
   352→                corrupted_image, manifest,
   353→                read_budget=read_budget,
   354→                corruption_metadata=corruption_meta,
   355→            )
   356→            metrics_carving = judge.judge(
   357→                recovered_files=[{"name": f.name, "sha256": f.sha256,
   358→                                  "size": f.size, "is_directory": f.is_directory}
   359→                                 for f in result_carving.recovered_files],
   360→                read_count=result_carving.read_count,
   361→                sectors_wasted=result_carving.sectors_wasted,
   362→                time_to_first_file=result_carving.time_to_first_file,
   363→                mft_entries_parsed=result_carving.mft_entries_parsed,
   364→            )
   365→
   366→            # ─── MFT-First ───────────────────────────────────────────
   367→            result_mft = motor_mft.recover(
   368→                corrupted_image, manifest,
   369→                read_budget=read_budget,
   370→                corruption_metadata=corruption_meta,
   371→            )
   372→            metrics_mft = judge.judge(
   373→                recovered_files=[{"name": f.name, "sha256": f.sha256,
   374→                                  "size": f.size, "is_directory": f.is_directory}
   375→                                 for f in result_mft.recovered_files],
   376→                read_count=result_mft.read_count,
   377→                sectors_wasted=result_mft.sectors_wasted,
   378→                time_to_first_file=result_mft.time_to_first_file,
   379→                mft_entries_parsed=result_mft.mft_entries_parsed,
   380→            )
   381→
   382→            # ─── Motor C ─────────────────────────────────────────────
   383→            result_motor_c = motor_c.recover(
   384→                corrupted_image, manifest,
   385→                read_budget=read_budget,
   386→                corruption_metadata=corruption_meta,
   387→            )
   388→            metrics_motor_c = judge.judge(
   389→                recovered_files=[{"name": f.name, "sha256": f.sha256,
   390→                                  "size": f.size, "is_directory": f.is_directory}
   391→                                 for f in result_motor_c.recovered_files],
   392→                read_count=result_motor_c.read_count,
   393→                sectors_wasted=result_motor_c.sectors_wasted,
   394→                time_to_first_file=result_motor_c.time_to_first_file,
   395→                mft_entries_parsed=result_motor_c.mft_entries_parsed,
   396→            )
   397→
   398→            # Collect recovery rates
   399→            carving_recoveries.append(metrics_carving.recovery_rate())
   400→            mft_recoveries.append(metrics_mft.recovery_rate())
   401→            motor_c_recoveries.append(metrics_motor_c.recovery_rate())
   402→
   403→            # Accumulate metrics (use last repetition for details)
   404→            if rep == n_repetitions - 1:
   405→                sum_carving = metrics_carving
   406→                sum_mft = metrics_mft
   407→                sum_motor_c = metrics_motor_c
   408→
   409→        # Compute statistics
   410→        stats = compute_statistics(carving_recoveries, mft_recoveries)
   411→
   412→        # Determine winner
   413→        avg_carving = sum(carving_recoveries) / len(carving_recoveries)
   414→        avg_mft = sum(mft_recoveries) / len(mft_recoveries)
   415→        avg_motor_c = sum(motor_c_recoveries) / len(motor_c_recoveries)
   416→
   417→        delta = avg_mft - avg_carving
   418→
   419→        if delta > 0.05:
   420→            winner = "MFT-First"
   421→        elif delta < -0.05:
   422→            winner = "Carving"
   423→        else:
   424→            winner = "Tie"
   425→
   426→        # Create data point
   427→        point = CrossoverPoint(
   428→            mft_damage_pct=damage_pct,
   429→            mft_confidence=confidence,
   430→            carving_recovery=avg_carving,
   431→            carving_correct=sum_carving.files_correct_checksum,
   432→            carving_corrupt=sum_carving.files_corrupt,
   433→            carving_false_positives=sum_carving.false_positives,
   434→            carving_reads=sum_carving.read_count,
   435→            carving_efficiency=sum_carving.read_efficiency(),
   436→            carving_integrity=sum_carving.integrity_score,
   437→            mft_recovery=avg_mft,
   438→            mft_correct=sum_mft.files_correct_checksum,
   439→            mft_corrupt=sum_mft.files_corrupt,
   440→            mft_false_positives=sum_mft.false_positives,
   441→            mft_reads=sum_mft.read_count,
   442→            mft_efficiency=sum_mft.read_efficiency(),
   443→            mft_integrity=sum_mft.integrity_score,
   444→            motor_c_recovery=avg_motor_c,
   445→            motor_c_correct=sum_motor_c.files_correct_checksum,
   446→            motor_c_corrupt=sum_motor_c.files_corrupt,
   447→            motor_c_false_positives=sum_motor_c.false_positives,
   448→            motor_c_reads=sum_motor_c.read_count,
   449→            motor_c_efficiency=sum_motor_c.read_efficiency(),
   450→            motor_c_integrity=sum_motor_c.integrity_score,
   451→            n_repetitions=n_repetitions,
   452→            carving_recovery_ci_lower=stats["ci_lower_carving"],
   453→            carving_recovery_ci_upper=stats["ci_upper_carving"],
   454→            mft_recovery_ci_lower=stats["ci_lower_mft"],
   455→            mft_recovery_ci_upper=stats["ci_upper_mft"],
   456→            p_value=stats["p_value"],
   457→            effect_size=stats["effect_size"],
   458→            optimal_strategy="mft_first" if delta > 0 else "carving",
   459→            delta_recovery=delta,
   460→        )
   461→        result.points.append(point)
   462→
   463→        # Print row
   464→        print(f"  {damage_pct:>7.0%} │ {confidence:>5.1%} │ "
   465→              f"{avg_carving:>9.1%} │ {avg_mft:>9.1%} │ "
   466→              f"{avg_motor_c:>9.1%} │ "
   467→              f"{delta:>+7.1%} │ {winner:>10s} │ "
   468→              f"{stats['p_value']:>7.3f} │ {stats['effect_size']:>+5.2f}")
   469→
   470→    # Find crossover
   471→    crossover = result.find_crossover()
   472→    print(f"{'─'*90}")
   473→    if crossover is not None:
   474→        print(f"  ★ CROSSOVER POINT: {crossover:.1%} MFT damage")
   475→        print(f"    Type: {result.crossover_type}")
   476→        print(f"    Below this point: MFT-First wins")
   477→        print(f"    Above this point: Carving wins")
   478→    else:
   479→        # Check if one strategy always wins
   480→        if all(p.delta_recovery > 0 for p in result.points):
   481→            print(f"  No crossover found — MFT-First wins at ALL damage levels")
   482→        elif all(p.delta_recovery < 0 for p in result.points):
   483→            print(f"  No crossover found — Carving wins at ALL damage levels")
   484→        else:
   485→            print(f"  Mixed results — no clear crossover point")
   486→
   487→    print(f"{'='*90}")
   488→
   489→    return result
