   111→def compute_result_hash(metrics_dict: Dict) -> str:
   112→    """Compute a deterministic hash of key metrics for bit-exact reproducibility check."""
   113→    hash_fields = {
   114→        "files_recovered": metrics_dict.get("files_recovered"),
   115→        "files_correct_checksum": metrics_dict.get("files_correct_checksum"),
   116→        "read_count": metrics_dict.get("read_count"),
   117→        "recovery_rate": metrics_dict.get("recovery_rate"),
   118→        "rvs": round(metrics_dict.get("rvs", 0), 6),
   119→        "weighted_functional_score": round(metrics_dict.get("weighted_functional_score", 0), 6),
   120→    }
   121→    hash_str = json.dumps(hash_fields, sort_keys=True)
   122→    return hashlib.sha256(hash_str.encode()).hexdigest()[:16]

   125→def run_single_execution(run_index: int, image: bytes, manifest: Dict,
   126→                         motor_name: str = "MFT-First") -> Dict:
   127→    """Run a single execution and return all measurements."""
   128→    if motor_name == "MFT-First":
   129→        motor = MotorBMFTFirst()
   130→    elif motor_name == "Carving":
   131→        motor = MotorCarving()
   132→    else:
   133→        raise ValueError(f"Unknown motor: {motor_name}")
   134→
   135→    judge = RecoveryJudge(manifest)
   136→
   137→    t_start = time.perf_counter()
   138→    result = motor.recover(image, manifest, read_budget=0)
   139→    t_end = time.perf_counter()
   140→    runtime_ms = (t_end - t_start) * 1000.0
   141→
   142→    judge_input = [{
   143→        "name": f.name,
   144→        "sha256": f.sha256,
   145→        "size": f.size,
   146→        "is_directory": f.is_directory,
   147→        "data": f.data,
   148→    } for f in result.recovered_files]
   149→
   150→    metrics = judge.judge(
   151→        recovered_files=judge_input,
   152→        read_count=result.read_count,
   153→        sectors_wasted=result.sectors_wasted,
   154→        time_to_first_file=result.time_to_first_file,
   155→        mft_entries_parsed=result.mft_entries_parsed,
   156→    )
   157→
   158→    utility = compute_overall_utility(metrics.rvs, metrics.weighted_functional_score)
   159→    metrics_dict = metrics.to_dict()
   160→
   161→    return {
   162→        "run": run_index,
   163→        "motor": motor_name,
   164→        "timestamp": datetime.datetime.now().isoformat(),
   165→        "overall_utility": utility["overall_utility"],
   166→        "rvs": metrics.rvs,
   167→        "fqs": metrics.weighted_functional_score,
   168→        "recovery_rate": metrics.recovery_rate(),
   169→        "read_count": metrics.read_count,
   170→        "runtime_ms": runtime_ms,
   171→        "result_hash": compute_result_hash(metrics_dict),
   172→        "files_recovered": metrics_dict.get("files_recovered", 0),
   173→        "files_correct_checksum": metrics_dict.get("files_correct_checksum", 0),
   174→        "files_missing": metrics_dict.get("files_missing", 0),
   175→        "integrity_score": metrics_dict.get("integrity_score", 0.0),
   176→        "read_efficiency": metrics.read_efficiency(),
   177→    }

   180→def analyze_seed_group(runs: List[Dict]) -> Dict:
   181→    """Analyze a single seed group (same as EXP-0001 analysis)."""
   182→    metrics_to_analyze = [
   183→        "overall_utility", "rvs", "fqs", "recovery_rate",
   184→        "read_count", "runtime_ms"
   185→    ]
   186→
   187→    summary = {}
   188→    for metric in metrics_to_analyze:
   189→        values = [r[metric] for r in runs]
   190→        mean = statistics.mean(values)
   191→        sd = statistics.stdev(values) if len(values) > 1 else 0.0
   192→        cv = (sd / mean * 100) if mean != 0 else 0.0
   193→        min_val = min(values)
   194→        max_val = max(values)
   195→
   196→        n = len(values)
   197→        if n > 1:
   198→            se = sd / (n ** 0.5)
   199→            t_val = 2.045 if n == 30 else 2.0
   200→            ci_lower = mean - t_val * se
   201→            ci_upper = mean + t_val * se
   202→        else:
   203→            ci_lower = ci_upper = mean
   204→
   205→        summary[metric] = {
   206→            "mean": round(mean, 6),
   207→            "sd": round(sd, 6),
   208→            "cv_percent": round(cv, 4),
   209→            "min": round(min_val, 6),
   210→            "max": round(max_val, 6),
   211→            "range": round(max_val - min_val, 6),
   212→            "ci_95_lower": round(ci_lower, 6),
   213→            "ci_95_upper": round(ci_upper, 6),
   214→            "n": n,
   215→        }
   216→
   217→    # Hash consistency
   218→    hashes = [r["result_hash"] for r in runs]
   219→    unique_hashes = set(hashes)
   220→    summary["hash_consistency"] = {
   221→        "unique_hashes": len(unique_hashes),
   222→        "all_identical": len(unique_hashes) == 1,
   223→        "hash_values": list(unique_hashes),
   224→    }
   225→
   226→    # Temporal drift
   227→    half = len(runs) // 2
   228→    first_half_ou = [r["overall_utility"] for r in runs[:half]]
   229→    second_half_ou = [r["overall_utility"] for r in runs[half:]]
   230→    drift = statistics.mean(second_half_ou) - statistics.mean(first_half_ou)
   231→    summary["temporal_drift"] = {
   232→        "first_half_mean": round(statistics.mean(first_half_ou), 6),
   233→        "second_half_mean": round(statistics.mean(second_half_ou), 6),
   234→        "drift": round(drift, 6),
   235→        "drift_percent": round(drift / statistics.mean(first_half_ou) * 100, 4) if statistics.mean(first_half_ou) != 0 else 0.0,
   236→        "interpretation": "No drift detected" if abs(drift) < 0.001 else "Potential drift detected",
   237→    }
   238→
   239→    return summary

   242→def analyze_cross_seed(seed_summaries: Dict[int, Dict], motor_name: str) -> Dict:
   243→    """Analyze cross-seed consistency: does the direction of CLAIM-001 hold?"""
   244→    ou_means = {}
   245→    for seed, summaries in seed_summaries.items():
   246→        if motor_name in summaries:
   247→            ou_means[seed] = summaries[motor_name]["overall_utility"]["mean"]
   248→
   249→    if len(ou_means) < 2:
   250→        return {"error": "Not enough seed groups for cross-seed analysis"}
   251→
   252→    values = list(ou_means.values())
   253→    cross_mean = statistics.mean(values)
   254→    cross_sd = statistics.stdev(values) if len(values) > 1 else 0.0
   255→    cross_cv = (cross_sd / cross_mean * 100) if cross_mean != 0 else 0.0
   256→
   257→    # Check if all seeds produce OU > 0 for MFT-First
   258→    all_positive = all(v > 0 for v in values)
   259→
   260→    # Check if direction is consistent
   261→    # For MFT-First: all should be > 0
   262→    # For Carving: all should be = 0 on healthy image
   263→    if motor_name == "MFT-First":
   264→        direction_consistent = all(v > 0.5 for v in values)
   265→    elif motor_name == "Carving":
   266→        direction_consistent = all(v == 0.0 for v in values)
   267→    else:
   268→        direction_consistent = False
   269→
   270→    return {
   271→        "cross_seed_mean": round(cross_mean, 6),
   272→        "cross_seed_sd": round(cross_sd, 6),
   273→        "cross_seed_cv_percent": round(cross_cv, 4),
   274→        "per_seed_means": {str(k): round(v, 6) for k, v in ou_means.items()},
   275→        "all_positive": all_positive,
   276→        "direction_consistent": direction_consistent,
   277→        "min_seed_mean": round(min(values), 6),
   278→        "max_seed_mean": round(max(values), 6),
   279→        "range": round(max(values) - min(values), 6),
   280→    }
