    61→    61→def add_styled_table(doc, headers, rows, col_widths=None):

    63→    63→    table.alignment = WD_TABLE_ALIGNMENT.CENTER
    64→    64→    table.style = 'Table Grid'

    67→    67→        cell.text = ""
    68→    68→        p = cell.paragraphs[0]
    69→    69→        run = p.add_run(header)
    70→    70→        run.bold = True
    71→    71→        run.font.size = Pt(10)
    72→    72→        run.font.color.rgb = c(P["white"])
    73→    73→        run.font.name = "Calibri"
    74→    74→        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    75→    75→        set_cell_shading(cell, P["primary"])
    76→    76→    for row_idx, row_data in enumerate(rows):
    77→    77→        for col_idx, cell_text in enumerate(row_data):
    78→    78→            cell = table.rows[row_idx + 1].cells[col_idx]
    79→    79→            cell.text = ""
    80→    80→            p = cell.paragraphs[0]
    81→    81→            run = p.add_run(str(cell_text))

    84→    84→            run.font.color.rgb = c(P["body"])

    86→    86→                set_cell_shading(cell, P["surface"])

    88→    88→        for row in table.rows:
    89→    89→            for idx, width in enumerate(col_widths):
