    50→    """A piece of evidence supporting or refuting a hypothesis."""
    51→    timestamp: str
    52→    type: EvidenceType
    53→    supports: bool                  # True = supports, False = refutes
    54→    description: str
    55→    experiment_id: str = ""         # Link to experiment results
    56→    strength: str = "moderate"      # weak | moderate | strong
    57→    details: Dict = field(default_factory=dict)
    58→
    59→
    60→@dataclass
    61→class Hypothesis:
    62→    """A single hypothesis with full tracking."""
    63→    id: str
    64→    statement: str
    65→    status: HypothesisStatus = HypothesisStatus.PENDING
    66→    confidence: float = 0.0
    67→    evidence: List[Evidence] = field(default_factory=list)
    68→    dependencies: List[str] = field(default_factory=list)
    69→    open_questions: List[str] = field(default_factory=list)
    70→    refinement_of: Optional[str] = None   # If this refines a previous hypothesis
    71→    created_at: str = ""
    72→    updated_at: str = ""
    73→
    74→    def add_evidence(self, evidence: Evidence):
    75→        """Add evidence and update status/confidence."""
    76→        self.evidence.append(evidence)
    77→        self.updated_at = datetime.now(timezone.utc).isoformat()
    78→        self._recompute_status()
    79→
    80→    def _recompute_status(self):
    81→        """Recompute status based on accumulated evidence."""
    82→        if not self.evidence:
    83→            self.status = HypothesisStatus.PENDING
    84→            return
    85→
    86→        supports = [e for e in self.evidence if e.supports]
    87→        refutes = [e for e in self.evidence if not e.supports]
    88→
    89→        support_strength = sum(
    90→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    91→            for e in supports
    92→        )
    93→        refute_strength = sum(
    94→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    95→            for e in refutes
    96→        )
    97→
    98→        total = support_strength + refute_strength
    99→        if total == 0:
   100→            self.confidence = 0.5
   101→            self.status = HypothesisStatus.INCONCLUSIVE
   102→            return
   103→
   104→        self.confidence = support_strength / total
   105→
   106→        if self.confidence >= 0.8:
   107→            self.status = HypothesisStatus.SUPPORTED
   108→        elif self.confidence <= 0.2:
   109→            self.status = HypothesisStatus.REFUTED
   110→        elif len(self.evidence) < 3:
   111→            self.status = HypothesisStatus.IN_EVALUATION
   112→        else:
   113→            self.status = HypothesisStatus.INCONCLUSIVE
   114→
   115→    def to_dict(self) -> Dict:
   116→        return {
   117→            "id": self.id,
   118→            "statement": self.statement,
   119→            "status": self.status.value,
   120→            "confidence": round(self.confidence, 3),
   121→            "evidence": [
   122→                {
   123→                    "timestamp": e.timestamp,
   124→                    "type": e.type.value,
   125→                    "supports": e.supports,
   126→                    "description": e.description,
   127→                    "experiment_id": e.experiment_id,
   128→                    "strength": e.strength,
   129→                }
   130→                for e in self.evidence
   131→            ],
   132→            "dependencies": self.dependencies,
   133→            "open_questions": self.open_questions,
   134→            "refinement_of": self.refinement_of,
   135→            "created_at": self.created_at,
   136→            "updated_at": self.updated_at,
   137→        }

   482→    def register(self, hypothesis_id: str, statement: str,
   483→                 status: HypothesisStatus = HypothesisStatus.PENDING,
   484→                 dependencies: List[str] = None,
   485→                 open_questions: List[str] = None,
   486→                 refinement_of: str = None) -> Hypothesis:
   487→        """Register a new hypothesis."""
   488→        if hypothesis_id in self.hypotheses:
   489→            return self.hypotheses[hypothesis_id]
   490→
   491→        now = datetime.now(timezone.utc).isoformat()
   492→        h = Hypothesis(
   493→            id=hypothesis_id,
   494→            statement=statement,
   495→            status=status,
   496→            dependencies=dependencies or [],
   497→            open_questions=open_questions or [],
   498→            refinement_of=refinement_of,
   499→            created_at=now,
   500→            updated_at=now,
   501→        )
   502→        self.hypotheses[hypothesis_id] = h
   503→        return h

   505→    def add_evidence(self, hypothesis_id: str, evidence: Evidence):
   506→        """Add evidence to a hypothesis."""
   507→        if hypothesis_id not in self.hypotheses:
   508→            raise ValueError(f"Unknown hypothesis: {hypothesis_id}")
   509→        self.hypotheses[hypothesis_id].add_evidence(evidence)

   511→    def get(self, hypothesis_id: str) -> Optional[Hypothesis]:
   512→        """Get a hypothesis by ID."""
   513→        return self.hypotheses.get(hypothesis_id)
   514→
   515→    def all_hypotheses(self) -> List[Hypothesis]:
   516→        """Get all hypotheses sorted by ID."""
   517→        return sorted(self.hypotheses.values(), key=lambda h: h.id)

   519→    def summary_table(self) -> str:
   520→        """Generate a markdown summary table of all hypotheses."""
   521→        lines = [
   522→            "# RecoveryLab — Hypothesis Registry",
   523→            "",
   524→            "| ID | Hypothesis | Status | Confidence | Evidence | Open Questions |",
   525→            "|----|-----------|--------|------------|----------|----------------|",
   526→        ]
   527→        for h in self.all_hypotheses():
   528→            supports = sum(1 for e in h.evidence if e.supports)
   529→            refutes = sum(1 for e in h.evidence if not e.supports)
   530→            evidence_str = f"{supports}S / {refutes}R"
   531→            questions = len(h.open_questions)
   532→            lines.append(
   533→                f"| {h.id} | {h.statement[:60]}... | {h.status.value} | "
   534→                f"{h.confidence:.0%} | {evidence_str} | {questions} |"

   544→            lines.append(f"**Status:** {h.status.value}")
   545→            lines.append(f"**Confidence:** {h.confidence:.0%}")
   546→            if h.refinement_of:
   547→                lines.append(f"**Refinement of:** {h.refinement_of}")
   548→            if h.dependencies:
   549→                lines.append(f"**Dependencies:** {', '.join(h.dependencies)}")
   550→            lines.append("")
   551→
   552→            if h.evidence:
   553→                lines.append("### Evidence")
   554→                for i, e in enumerate(h.evidence, 1):
   555→                    symbol = "✓" if e.supports else "✗"
   556→                    lines.append(
   557→                        f"{i}. {symbol} [{e.type.value}] {e.description} "
   558→                        f"(strength: {e.strength})"
   559→                    )
   560→                lines.append("")
   561→
   562→            if h.open_questions:
   563→                lines.append("### Open Questions")
   564→                for q in h.open_questions:
   565→                    lines.append(f"- {q}")
   566→                lines.append("")
   567→
   568→            lines.append("---")
   569→            lines.append("")
   570→
   571→        return "\n".join(lines)
   572→
   573→    def save(self, path: Path):
   574→        """Save the registry to a JSON file."""
   575→        data = {
   576→            "generated_at": datetime.now(timezone.utc).isoformat(),
   577→            "hypotheses": {h_id: h.to_dict() for h_id, h in self.hypotheses.items()},
   578→        }
   579→        path.parent.mkdir(parents=True, exist_ok=True)
   580→        with open(path, 'w') as f:
   581→            json.dump(data, f, indent=2, ensure_ascii=False)
   582→
   583→    def load(self, path: Path):
   584→        """Load the registry from a JSON file."""
   585→        with open(path, 'r') as f:
   586→            data = json.load(f)
   587→
   588→        for h_id, h_data in data.get("hypotheses", {}).items():
   589→            if h_id in self.hypotheses:
   590→                # Update existing hypothesis with loaded evidence
   591→                h = self.hypotheses[h_id]
   592→                for e_data in h_data.get("evidence", []):
   593→                    # Check if we already have this evidence
   594→                    already_exists = any(
   595→                        e.description == e_data["description"]
   596→                        for e in h.evidence
   597→                    )
   598→                    if not already_exists:
   599→                        h.add_evidence(Evidence(
   600→                            timestamp=e_data["timestamp"],
   601→                            type=EvidenceType(e_data["type"]),
   602→                            supports=e_data["supports"],
   603→                            description=e_data["description"],
   604→                            experiment_id=e_data.get("experiment_id", ""),
   605→                            strength=e_data.get("strength", "moderate"),
   606→                        ))
   607→
   608→    def next_steps(self) -> List[str]:
   609→        """What experiments should we run next to close open hypotheses?"""
   610→        steps = []
   611→
   612→        # Check what's most needed
   613→        for h in self.all_hypotheses():
   614→            if h.status == HypothesisStatus.PENDING:
   615→                steps.append(
   616→                    f"[{h.id}] PENDING — needs first experiment: {h.open_questions[0] if h.open_questions else 'design experiment'}"
   617→                )
   618→            elif h.status == HypothesisStatus.IN_EVALUATION:
   619→                steps.append(
   620→                    f"[{h.id}] IN EVALUATION — needs more evidence: {h.open_questions[0] if h.open_questions else 'gather more data'}"
   621→                )
   622→            elif h.status == HypothesisStatus.INCONCLUSIVE:
   623→                steps.append(
   624→                    f"[{h.id}] INCONCLUSIVE — conflicting evidence, need decisive experiment"
   625→                )
   626→
   627→        return steps
