    99→    99→def compute_result_hash(metrics_dict: Dict) -> str:
   100→   100→    """Compute a deterministic hash of key metrics for bit-exact reproducibility check."""
   101→   101→    # Only hash the deterministic parts: counts, rates, not timestamps
   102→   102→    hash_fields = {
   103→   103→        "files_recovered": metrics_dict.get("files_recovered"),
   104→   104→        "files_correct_checksum": metrics_dict.get("files_correct_checksum"),
   105→   105→        "read_count": metrics_dict.get("read_count"),
   106→   106→        "recovery_rate": metrics_dict.get("recovery_rate"),
   107→   107→        "rvs": round(metrics_dict.get("rvs", 0), 6),
   108→   108→        "weighted_functional_score": round(metrics_dict.get("weighted_functional_score", 0), 6),
   109→   109→    }
   110→   110→    hash_str = json.dumps(hash_fields, sort_keys=True)
   111→   111→    return hashlib.sha256(hash_str.encode()).hexdigest()[:16]

   114→   114→def run_single_execution(run_index: int, image: bytes, manifest: Dict, motor_name: str = "MFT-First") -> Dict:
   115→   115→    """Run a single baseline execution and return all measurements."""
   116→   116→    if motor_name == "MFT-First":
   117→   117→        motor = MotorBMFTFirst()
   118→   118→    elif motor_name == "Carving":
   119→   119→        motor = MotorCarving()
   120→   120→    else:
   121→   121→        raise ValueError(f"Unknown motor: {motor_name}")
   122→   122→    judge = RecoveryJudge(manifest)
   123→   123→
   124→   124→    # ── Execute recovery ──
   125→   125→    t_start = time.perf_counter()
   126→   126→    result = motor.recover(image, manifest, read_budget=0)
   127→   127→    t_end = time.perf_counter()
   128→   128→    runtime_ms = (t_end - t_start) * 1000.0
   129→   129→
   130→   130→    # ── Convert to Judge format ──
   131→   131→    judge_input = [{
   132→   132→        "name": f.name,
   133→   133→        "sha256": f.sha256,
   134→   134→        "size": f.size,
   135→   135→        "is_directory": f.is_directory,
   136→   136→        "data": f.data,
   137→   137→    } for f in result.recovered_files]
   138→   138→
   139→   139→    # ── Judge evaluation ──
   140→   140→    metrics = judge.judge(
   141→   141→        recovered_files=judge_input,
   142→   142→        read_count=result.read_count,
   143→   143→        sectors_wasted=result.sectors_wasted,
   144→   144→        time_to_first_file=result.time_to_first_file,
   145→   145→        mft_entries_parsed=result.mft_entries_parsed,
   146→   146→    )
   147→   147→
   148→   148→    # ── Compute Overall Utility ──
   149→   149→    utility = compute_overall_utility(metrics.rvs, metrics.weighted_functional_score)
   150→   150→
   151→   151→    # ── Build metrics dict ──
   152→   152→    metrics_dict = metrics.to_dict()
   153→   153→
   154→   154→    return {
   155→   155→        "run": run_index,
   156→   156→        "motor": motor_name,
   157→   157→        "timestamp": datetime.datetime.now().isoformat(),
   158→   158→        "overall_utility": utility["overall_utility"],
   159→   159→        "rvs": metrics.rvs,
   160→   160→        "fqs": metrics.weighted_functional_score,
   161→   161→        "recovery_rate": metrics.recovery_rate(),
   162→   162→        "read_count": metrics.read_count,
   163→   163→        "runtime_ms": runtime_ms,
   164→   164→        "result_hash": compute_result_hash(metrics_dict),
   165→   165→        "files_recovered": metrics_dict.get("files_recovered", 0),
   166→   166→        "files_correct_checksum": metrics_dict.get("files_correct_checksum", 0),
   167→   167→        "files_missing": metrics_dict.get("files_missing", 0),
   168→   168→        "integrity_score": metrics_dict.get("integrity_score", 0.0),
   169→   169→        "read_efficiency": metrics.read_efficiency(),
   170→   170→        "mft_entries_parsed": metrics.mft_entries_parsed,
   171→   171→        "sectors_wasted": metrics.sectors_wasted,
   172→   172→        "time_to_first_file": metrics.time_to_first_file,
   173→   173→    }

   176→   176→def analyze_results(runs: List[Dict]) -> Dict:
   177→   177→    """Compute summary statistics for all measured metrics."""
   178→   178→    metrics_to_analyze = [
   179→   179→        "overall_utility", "rvs", "fqs", "recovery_rate",
   180→   180→        "read_count", "runtime_ms"
   181→   181→    ]
   182→   182→
   183→   183→    summary = {}
   184→   184→    for metric in metrics_to_analyze:
   185→   185→        values = [r[metric] for r in runs]
   186→   186→        mean = statistics.mean(values)
   187→   187→        sd = statistics.stdev(values) if len(values) > 1 else 0.0
   188→   188→        cv = (sd / mean * 100) if mean != 0 else 0.0
   189→   189→        min_val = min(values)
   190→   190→        max_val = max(values)
   191→   191→
   192→   192→        # 95% confidence interval (t-distribution approximation for small samples)
   193→   193→        n = len(values)
   194→   194→        if n > 1:
   195→   195→            se = sd / (n ** 0.5)
   196→   196→            # t-value for 95% CI, df=n-1 (approximate with 2.045 for n=30)
   197→   197→            t_val = 2.045 if n == 30 else 2.0  # approximate
   198→   198→            ci_lower = mean - t_val * se
   199→   199→            ci_upper = mean + t_val * se
   200→   200→        else:
   201→   201→            ci_lower = ci_upper = mean
   202→   202→
   203→   203→        summary[metric] = {
   204→   204→            "mean": round(mean, 6),
   205→   205→            "sd": round(sd, 6),
   206→   206→            "cv_percent": round(cv, 4),
   207→   207→            "min": round(min_val, 6),
   208→   208→            "max": round(max_val, 6),
   209→   209→            "range": round(max_val - min_val, 6),
   210→   210→            "ci_95_lower": round(ci_lower, 6),
   211→   211→            "ci_95_upper": round(ci_upper, 6),
   212→   212→            "n": n,
   213→   213→        }
   214→   214→
   215→   215→    # ── Hash consistency check ──
   216→   216→    hashes = [r["result_hash"] for r in runs]
   217→   217→    unique_hashes = set(hashes)
   218→   218→    summary["hash_consistency"] = {
   219→   219→        "unique_hashes": len(unique_hashes),
   220→   220→        "all_identical": len(unique_hashes) == 1,
   221→   221→        "hash_values": list(unique_hashes),
   222→   222→    }
   223→   223→
   224→   224→    # ── Temporal drift check ──
   225→   225→    # Split runs into first half and second half, compare means
   226→   226→    half = len(runs) // 2
   227→   227→    first_half_ou = [r["overall_utility"] for r in runs[:half]]
   228→   228→    second_half_ou = [r["overall_utility"] for r in runs[half:]]
   229→   229→    drift = statistics.mean(second_half_ou) - statistics.mean(first_half_ou)
   230→   230→    summary["temporal_drift"] = {
   231→   231→        "first_half_mean": round(statistics.mean(first_half_ou), 6),
   232→   232→        "second_half_mean": round(statistics.mean(second_half_ou), 6),
   233→   233→        "drift": round(drift, 6),
   234→   234→        "drift_percent": round(drift / statistics.mean(first_half_ou) * 100, 4) if statistics.mean(first_half_ou) != 0 else 0.0,
   235→   235→        "interpretation": "No drift detected" if abs(drift) < 0.001 else "Potential drift detected — investigate",
   236→   236→    }
   237→   237→
   238→   238→    # ── Success criteria evaluation ──
   239→   239→    criteria = {
   240→   240→        "30_executions_completed": len(runs) == 30,
   241→   241→        "no_errors": True,  # If we got here, no errors
   242→   242→        "hash_identical": len(unique_hashes) == 1,
   243→   243→        "no_temporal_drift": abs(drift) < 0.001,
   244→   244→        "cv_overall_utility": summary["overall_utility"]["cv_percent"],
   245→   245→        "cv_note": "CV threshold (X) to be defined after data collection",
   246→   246→    }
   247→   247→    summary["success_criteria"] = criteria
   248→   248→
   249→   249→    # ── Empirical threshold calculation (ED-008) ──
   250→   250→    ou_values = [r["overall_utility"] for r in runs]
   251→   251→    ou_sd = statistics.stdev(ou_values) if len(ou_values) > 1 else 0.0
   252→   252→    threshold = max(2 * ou_sd, 0.01)
   253→   253→    summary["empirical_threshold"] = {
   254→   254→        "overall_utility_sd": round(ou_sd, 6),
   255→   255→        "threshold_2sigma": round(threshold, 6),
   256→   256→        "threshold_percent": round(threshold * 100, 4),
   257→   257→        "interpretation": f"A difference in Overall Utility must exceed {threshold:.4f} ({threshold*100:.2f}%) to be considered significant at 2-sigma",
   258→   258→    }
   259→   259→
   260→   260→    return summary
