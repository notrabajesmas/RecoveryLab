     4→BLOCKER-001 resolution: a TRUE carving strategy that NEVER touches the MFT.

     6→Philosophy:
     7→  - Uses ONLY file signatures (magic bytes) to identify files

   301→
   302→class MotorCarving(BaseMotor):
   303→    """
   304→    Motor Carving: signature-based file recovery.
   305→
   306→    This motor NEVER reads the MFT. It scans the raw disk image
   307→    byte-by-byte looking for known file signatures.
   308→
   309→    Strategy:
   310→      1. Scan the ENTIRE image for file signatures
   311→      2. When a signature is found, extract the file
   312→      3. Use footer markers to determine file boundaries
   313→      4. If no footer, use max-size heuristic
   314→      5. Assign generic names (carved_0001.jpg, etc.)
   315→
   316→    This is the "adversarial baseline" — it represents the best
   317→    you can do without any filesystem metadata.
   318→    """
   319→
   320→    @property
   321→    def name(self) -> str:
   322→        return "Motor Carving (Signature-Only)"
   323→
   324→    @property
   325→    def description(self) -> str:
   326→        return ("Carving: scans raw bytes for file signatures. "
   327→                "NEVER reads MFT, VBR, or any filesystem metadata. "
   328→                "No filenames, no directories, no fragmentation support.")
   329→
   330→    def recover(self, image: bytes, manifest: Dict,
   331→                read_budget: int = 0,
   332→                corruption_metadata: Optional[Dict] = None) -> MotorResult:
   333→        """
   334→        Run signature-based carving recovery.
   335→
   336→        Key constraint: This method NEVER accesses:
   337→          - manifest["mft"] (MFT data)
   338→          - manifest["vbr"] (VBR data)
   339→          - Any filesystem metadata
   340→
   341→        It ONLY uses manifest["cluster_size"] and manifest["total_clusters"]
   342→        to understand the image geometry, and manifest["files"] to know
   343→        what file types exist (for reporting purposes only — NOT for recovery).
   344→
   345→        Actually, to be STRICTLY pure carving, we should NOT use the manifest
   346→        at all for recovery. The manifest is only used for Judge evaluation
   347→        after recovery.
   348→        """
   349→        result = MotorResult(motor_name=self.name)
   350→        cluster_size = manifest.get("cluster_size", 4096)
   351→        total_clusters = manifest.get("total_clusters", len(image) // cluster_size)
   352→
   353→        reads = 0
   354→        sectors_per_cluster = cluster_size // 512
   355→        first_file_reads = 0
   356→        found_first_file = False
   357→
   358→        # ─── Phase 1: Scan entire image for signatures ────────────────
   359→        # We read every cluster sequentially (like a real carver would)
   360→        # This is the MAXIMUM read cost — no metadata guidance at all
   361→
   362→        carved_files: List[Dict] = []
   363→        signatures_found: Dict[str, int] = {}
   364→        scan_offset = 0
   365→
   366→        # Build a list of all header patterns to search for
   367→        # We scan byte-by-byte through the image
   368→        image_len = len(image)
   369→
   370→        # Track which clusters we've already read (for read counting)
   371→        read_clusters: Set[int] = set()
   372→
   373→        # Scan the image looking for file signatures
   374→        # We scan at every byte position within the first sector of each cluster
   375→        # (in practice, signatures are usually at the start of a cluster)
   376→        for cluster_num in range(total_clusters):
   377→            cluster_start = cluster_num * cluster_size
   378→
   379→            if cluster_start + cluster_size > image_len:
   380→                break
   381→
   382→            # Read this cluster
   383→            cluster_data = image[cluster_start:cluster_start + cluster_size]
   384→            read_clusters.add(cluster_num)
   385→            reads += sectors_per_cluster
   386→
   387→            # Check for budget exhaustion
   388→            if read_budget > 0 and reads >= read_budget:
   389→                break
   390→
   391→            # Search for signatures within this cluster
   392→            # Check at the start of the cluster (most common case)
   393→            # and also at every offset within the first 512 bytes
   394→            # (some signatures might be offset within a cluster)
   395→            for sig in SIGNATURES:
   396→                # Quick check: does the signature match at the start?
   397→                matches = self._find_signature_matches(
   398→                    image, cluster_start, cluster_size, sig, image_len
   399→                )
   400→
   401→                for match_offset in matches:
   402→                    # Found a signature! Try to extract the file
   403→                    carved = self._carve_file(
   404→                        image, match_offset, sig, image_len,
   405→                        cluster_size, total_clusters, read_clusters
   406→                    )
   407→
   408→                    if carved is not None:
   409→                        carved_files.append(carved)
   410→                        sig_name = sig.name
   411→                        signatures_found[sig_name] = signatures_found.get(sig_name, 0) + 1
   412→
   413→                        # Update read count for any additional clusters we read
   414→                        # beyond the current scan position
   415→                        extra_reads = carved.get("extra_clusters_read", 0)
   416→                        reads += extra_reads * sectors_per_cluster
   417→
   418→        # ─── Phase 2: Resolve ZIP/DOCX ambiguity ─────────────────────
   419→        # ZIP and DOCX share the same PK\x03\x04 header.
   420→        # We try to distinguish them by looking for internal structure.
   421→        carved_files = self._resolve_zip_docx(carved_files)
   422→
   423→        # ─── Phase 3: Deduplicate overlapping carves ──────────────────
   424→        # If two signatures overlap (e.g., ZIP and DOCX at same offset),
   425→        # keep the more specific one
   426→        carved_files = self._deduplicate_carves(carved_files)
   427→
   428→        # ─── Phase 4: Build result ────────────────────────────────────
   429→        # Assign generic names
   430→        type_counters: Dict[str, int] = {}
   431→
   432→        for carved in carved_files:
   433→            ext = carved["extension"]
   434→            type_counters[ext] = type_counters.get(ext, 0) + 1
   435→            name = f"carved_{type_counters[ext]:04d}{ext}"
   436→
   437→            data = carved["data"]
   438→            sha256 = hashlib.sha256(data).hexdigest()
   439→
   440→            if not found_first_file:
   441→                first_file_reads = reads
   442→                found_first_file = True
   443→
   444→            result.recovered_files.append(RecoveredFile(
   445→                name=name,
   446→                sha256=sha256,
   447→                size=len(data),
   448→                data=data,
   449→                source="carving",  # Mark as carving-recovered
   450→                read_count=reads,
   451→            ))
   452→
   453→        # ─── Compute final metrics ────────────────────────────────────
   454→        result.read_count = reads
   455→        result.time_to_first_file = first_file_reads
   456→        result.mft_entries_parsed = 0  # NEVER parses MFT
   457→        result.total_time_seconds = reads * 0.001
   458→        result.directories_rebuilt = 0  # No directory support
   459→
   460→        # Sectors wasted: we read everything, but not all clusters had files
   461→        # In carving, "wasted" is relative — we had to read everything to find files
   462→        useful_clusters = len(set(
   463→            cf.get("start_cluster", 0) // cluster_size
   464→            for cf in carved_files
   465→        ))
   466→        result.sectors_wasted = max(0, reads - useful_clusters * sectors_per_cluster)
   467→
   468→        # Store carving statistics for debugging
   469→        result.carving_stats = {
   470→            "signatures_found": signatures_found,
   471→            "files_carved": len(carved_files),
   472→            "total_clusters_scanned": len(read_clusters),
   473→            "scan_coverage_pct": len(read_clusters) / total_clusters if total_clusters else 0,
   474→        }
   475→
   476→        return result
   477→
   478→    def _find_signature_matches(self, image: bytes, cluster_start: int,
   479→                                 cluster_size: int, sig: FileSignature,
   480→                                 image_len: int) -> List[int]:
   481→        """
   482→        Find all occurrences of a signature within a cluster.
   483→
   484→        Returns list of absolute offsets where the signature was found.
   485→        """
   486→        matches = []
   487→
   488→        # Check at the start of the cluster (most common)
   489→        # and at a few key offsets within the first 512 bytes
   490→        check_offsets = [0]
   491→
   492→        # Also check at sector-aligned offsets within the cluster
   493→        for sector_offset in range(512, min(cluster_size, 4096), 512):
   494→            check_offsets.append(sector_offset)
   495→
   496→        for offset_in_cluster in check_offsets:
   497→            abs_offset = cluster_start + offset_in_cluster
   498→            if abs_offset + sig.header_len > image_len:
   499→                continue
   500→
   501→            # Check if header matches (with mask)
   502→            candidate = image[abs_offset:abs_offset + sig.header_len]
   503→            if self._matches_with_mask(candidate, sig.header, sig.header_mask):
   504→                matches.append(abs_offset)
   505→
   506→        return matches
   507→
   508→    def _matches_with_mask(self, candidate: bytes, header: bytes,
   509→                            mask: bytes) -> bool:
   510→        """Check if candidate bytes match the header pattern with mask."""
   511→        if len(candidate) < len(header):
   512→            return False
   513→        for i in range(len(header)):
   514→            if mask[i] == 0xFF:  # Must match
   515→                if candidate[i] != header[i]:
   516→                    return False
   517→            # 0x00 = don't care, skip
   518→        return True
   519→
   520→    def _carve_file(self, image: bytes, start_offset: int,
   521→                     sig: FileSignature, image_len: int,
   522→                     cluster_size: int, total_clusters: int,
   523→                     read_clusters: Set[int]) -> Optional[Dict]:
   524→        """
   525→        Extract a file from the image starting at the given offset.
   526→
   527→        Uses footer markers to determine file boundaries.
   528→        If no footer is found, uses max-size heuristic.
   529→        """
   530→        # Check minimum size
   531→        remaining = image_len - start_offset
   532→        if remaining < sig.min_size:
   533→            return None
   534→
   535→        # Determine file end
   536→        file_data = None
   537→        extra_clusters_read = 0
   538→
   539→        if sig.name == "JPEG":
   540→            # JPEG-specific carving: parse structure to find the real EOI.
   541→            # RC-002: simple footer search (first or last FFD9) fails because
   542→            # the JPEG body can contain spurious FFD9 bytes. Structural parsing
   543→            # correctly identifies the EOI by skipping entropy-coded data.
   544→            #
   545→            # _carve_jpeg() returns a fully-formed result dict (with "data",
   546→            # "start_offset", etc.) — NOT raw bytes. We return it directly
   547→            # because the size validation is already done inside _carve_jpeg.
   548→            jpeg_result = self._carve_jpeg(
   549→                image, start_offset, image_len, sig, cluster_size,
   550→                total_clusters, read_clusters
   551→            )
   552→            if jpeg_result is not None:
   553→                return jpeg_result  # Already validated inside _carve_jpeg
   554→            return None
   555→        elif sig.footer:
   556→            # Search for footer after the header
   557→            # Start searching from after the header
   558→            footer_search_start = start_offset + sig.header_len
   559→            max_search_end = min(start_offset + sig.max_size, image_len)
   560→
   561→            footer_offset = self._find_footer(
   562→                image, footer_search_start, max_search_end, sig.footer
   563→            )
   564→
   565→            if footer_offset is not None:
   566→                # Found footer — extract file including footer
   567→                file_end = footer_offset + len(sig.footer)
   568→                file_data = image[start_offset:file_end]
   569→
   570→                # Track additional clusters read beyond the scan
   571→                end_cluster = (file_end - 1) // cluster_size
   572→                start_cluster = start_offset // cluster_size
   573→                for c in range(start_cluster, end_cluster + 1):
   574→                    if c not in read_clusters:
   575→                        read_clusters.add(c)
   576→                        extra_clusters_read += 1
   577→            else:
   578→                # No footer found — use max-size heuristic
   579→                # But cap at a reasonable size
   580→                heuristic_size = min(sig.max_size, remaining)
   581→                file_data = image[start_offset:start_offset + heuristic_size]
   582→        else:
   583→            # No footer defined — use max-size heuristic
   584→            # For MP4, try to parse the ftyp box to get the size
   585→            if sig.name == "MP4":
   586→                file_data = self._carve_mp4(image, start_offset, image_len, sig)
   587→            else:
   588→                heuristic_size = min(sig.max_size, remaining)
   589→                file_data = image[start_offset:start_offset + heuristic_size]
   590→
   591→        if file_data is None or len(file_data) < sig.min_size:
   592→            return None
   593→
   594→        # Validate: check if file is too large (likely a false positive)
   595→        if len(file_data) > sig.max_size:
   596→            file_data = file_data[:sig.max_size]
   597→
   598→        return {
   599→            "data": file_data,
   600→            "start_offset": start_offset,
   601→            "start_cluster": start_offset // cluster_size,
   602→            "extension": sig.extension,
   603→            "format_name": sig.name,
   604→            "extra_clusters_read": extra_clusters_read,
   605→            "size": len(file_data),
   606→        }

   608→    def _carve_jpeg(self, image: bytes, start_offset: int,
   609→                     image_len: int, sig: FileSignature,
   610→                     cluster_size: int, total_clusters: int,
   611→                     read_clusters: Set[int]) -> Optional[Dict]:
   612→        """
   613→        Carve a JPEG file by parsing its structure to find the real EOI.
   614→        
   615→        RC-002 fix: three-tier strategy:
   616→          1. Structural parsing (for real JPEGs with SOS marker)
   617→          2. Last FFD9 before next JPEG signature (for synthetic/partial JPEGs)
   618→          3. Last FFD9 within max_size (fallback if no other JPEGs nearby)
   619→        
   620→        The JPEG body can contain spurious FFD9 bytes (EXIF thumbnails,
   621→        random byte coincidences in entropy data). Structural parsing
   622→        correctly identifies EOI by skipping entropy-coded data (byte stuffing).
   623→        For synthetic JPEGs without proper markers, we use the next JPEG
   624→        signature as a boundary to avoid over-extension.
   625→        """
   626→        max_end = min(start_offset + sig.max_size, image_len)
   627→        
   628→        # Verify SOI
   629→        if start_offset + 2 > image_len:
   630→            return None
   631→        if image[start_offset:start_offset + 2] != b'\xFF\xD8':
   632→            return None
   633→        
   634→        # ─── Tier 1: Last FFD9 before next JPEG signature ──────────────
   635→        # For most cases (multiple JPEGs in the image), this is the most
   636→        # reliable approach. The next JPEG signature acts as a natural boundary.
   637→        next_jpeg = self._find_next_jpeg_signature(
   638→            image, start_offset + 4, max_end, cluster_size
   639→        )
   640→        
   641→        if next_jpeg is not None:
   642→            # Search for the last FFD9 before the next JPEG
   643→            search_end = next_jpeg
   644→            last_ffd9 = self._find_footer_last(
   645→                image, start_offset + 4, search_end, b'\xFF\xD9'
   646→            )
   647→            if last_ffd9 is not None:
   648→                file_end = last_ffd9 + 2
   649→                return self._build_jpeg_result(
   650→                    image, start_offset, file_end, sig, cluster_size,
   651→                    total_clusters, read_clusters
   652→                )
   653→        
   654→        # ─── Tier 2: Structural parsing (for real JPEGs with SOS) ──────
   655→        # Only used when there's no next JPEG boundary. This works for
   656→        # real JPEGs with proper markers (SOS, byte stuffing).
   657→        # Not used first because synthetic JPEGs have random FF bytes
   658→        # that create false SOS markers.
   659→        eoi_offset = self._find_jpeg_eoi_structural(image, start_offset, max_end)
   660→        
   661→        if eoi_offset is not None:
   662→            file_end = eoi_offset + 2  # Include the FFD9 bytes
   663→            return self._build_jpeg_result(
   664→                image, start_offset, file_end, sig, cluster_size,
   665→                total_clusters, read_clusters
   666→            )
   667→        
   668→        # ─── Tier 3: Last FFD9 within max_size ─────────────────────────
   669→        # Fallback: no structural parsing success, no next JPEG boundary.
   670→        # Use the last FFD9 within max_size.
   671→        last_ffd9 = self._find_footer_last(
   672→            image, start_offset + 4, max_end, b'\xFF\xD9'
   673→        )
   674→        if last_ffd9 is not None:
   675→            file_end = last_ffd9 + 2
   676→            return self._build_jpeg_result(
   677→                image, start_offset, file_end, sig, cluster_size,
   678→                total_clusters, read_clusters
   679→            )
   680→        
   681→        # No EOI found — use max-size heuristic
   682→        remaining = min(sig.max_size, image_len - start_offset)
   683→        file_data = image[start_offset:start_offset + remaining]
   684→        
   685→        if len(file_data) < sig.min_size:
   686→            return None
   687→        
   688→        return {
   689→            "data": file_data,
   690→            "start_offset": start_offset,
   691→            "start_cluster": start_offset // cluster_size,
   692→            "extension": sig.extension,
   693→            "format_name": sig.name,
   694→            "extra_clusters_read": 0,
   695→            "size": len(file_data),
   696→        }

   698→    def _find_jpeg_eoi_structural(self, image: bytes, start_offset: int,
   699→                                   max_end: int) -> Optional[int]:
   700→        """
   701→        Find the real EOI of a JPEG by parsing its structure.
   702→        
   703→        Returns the offset of the FFD9 marker (not including the marker bytes),
   704→        or None if structural parsing fails.
   705→        """
   706→        pos = start_offset + 2  # Skip SOI
   707→        
   708→        while pos < max_end - 1:
   709→            if image[pos] != 0xFF:
   710→                pos += 1
   711→                continue
   712→            
   713→            marker_byte = image[pos + 1]
   714→            
   715→            # FF 00 = byte stuffing in entropy data (skip)
   716→            # FF FF = padding (skip)
   717→            if marker_byte == 0x00 or marker_byte == 0xFF:
   718→                pos += 1
   719→                continue
   720→            
   721→            # SOS marker (FF DA) — start of entropy-coded data
   722→            if marker_byte == 0xDA:
   723→                if pos + 4 > max_end:
   724→                    return None
   725→                sos_length = struct.unpack('>H', image[pos + 2:pos + 4])[0]
   726→                pos += 2 + sos_length
   727→                
   728→                # Scan entropy data for real EOI
   729→                while pos < max_end - 1:
   730→                    if image[pos] == 0xFF:
   731→                        next_byte = image[pos + 1]
   732→                        if next_byte == 0x00:
   733→                            pos += 2
   734→                            continue
   735→                        if next_byte == 0xFF:
   736→                            pos += 1
   737→                            continue
   738→                        if next_byte == 0xD9:
   739→                            return pos  # Found real EOI
   740→                        # RST markers (D0-D7) — continue in entropy
   741→                        if 0xD0 <= next_byte <= 0xD7:
   742→                            pos += 2
   743→                            continue
   744→                        # Other marker — might be after entropy data
   745→                        pos += 1
   746→                        continue
   747→                    else:
   748→                        pos += 1
   749→                return None  # Ran out of data
   750→            
   751→            # EOI before SOS (unusual but valid)
   752→            if marker_byte == 0xD9:
   753→                return pos
   754→            
   755→            # RST markers — no length field
   756→            if marker_byte in (0xD0, 0xD1, 0xD2, 0xD3, 0xD4, 0xD5, 0xD6, 0xD7):
   757→                pos += 2
   758→                continue
   759→            
   760→            # Other markers with length field
   761→            if pos + 4 > max_end:
   762→                return None
   763→            marker_length = struct.unpack('>H', image[pos + 2:pos + 4])[0]
   764→            pos += 2 + marker_length
   765→        
   766→        return None  # Structural parsing failed

   768→    def _find_next_jpeg_signature(self, image: bytes, start: int,
   769→                                    end: int, cluster_size: int = 4096) -> Optional[int]:
   770→        """
   771→        Find the next JPEG signature (FFD8FF) after the given offset.
   772→        
   773→        Used as a boundary for JPEG carving: the last FFD9 before the
   774→        next JPEG is likely the real EOI of the current file.
   775→        
   776→        IMPORTANT: Only accepts signatures at cluster boundaries to avoid
   777→        false positives from random FFD8FF sequences in entropy data.
   778→        In NTFS, files start at cluster boundaries, so a real JPEG signature
   779→        should be aligned to a cluster boundary.
   780→        """
   781→        jpeg_sig = b'\xFF\xD8\xFF'
   782→        pos = start
   783→        
   784→        while pos < end - len(jpeg_sig):
   785→            idx = image.find(jpeg_sig, pos, end)
   786→            if idx == -1:
   787→                return None
   788→            
   789→            # Only accept signatures at cluster boundaries
   790→            # This filters out false FFD8FF sequences in random data
   791→            if idx % cluster_size == 0:
   792→                return idx
   793→            
   794→            # Skip to the next cluster boundary
   795→            next_cluster = ((idx // cluster_size) + 1) * cluster_size
   796→            pos = next_cluster
   797→        
   798→        return None

   800→    def _build_jpeg_result(self, image: bytes, start_offset: int,
   801→                            file_end: int, sig: FileSignature,
   802→                            cluster_size: int, total_clusters: int,
   803→                            read_clusters: Set[int]) -> Optional[Dict]:
   804→        """Build a JPEG carving result dict."""
   805→        file_data = image[start_offset:file_end]
   806→        
   807→        if len(file_data) < sig.min_size:
   808→            return None
   809→        
   810→        if len(file_data) > sig.max_size:
   811→            file_data = file_data[:sig.max_size]
   812→        
   813→        extra_clusters_read = 0
   814→        end_cluster = (file_end - 1) // cluster_size
   815→        start_cluster = start_offset // cluster_size
   816→        for c in range(start_cluster, end_cluster + 1):
   817→            if c not in read_clusters:
   818→                read_clusters.add(c)
   819→                extra_clusters_read += 1
   820→        
   821→        return {
   822→            "data": file_data,
   823→            "start_offset": start_offset,
   824→            "start_cluster": start_offset // cluster_size,
   825→            "extension": sig.extension,
   826→            "format_name": sig.name,
   827→            "extra_clusters_read": extra_clusters_read,
   828→            "size": len(file_data),
   829→        }

   831→    def _find_footer(self, image: bytes, start: int, end: int,
   832→                      footer: bytes) -> Optional[int]:
   833→        """
   834→        Find the first occurrence of footer bytes in the image.
   835→
   836→        Uses a simple scan. For large images, this could be optimized
   837→        with a Boyer-Moore search, but for our 10MB images it's fine.
   838→        """
   839→        footer_len = len(footer)
   840→        if footer_len == 0:
   841→            return None
   842→
   843→        # Search in chunks to avoid excessive memory usage
   844→        chunk_size = 1024 * 1024  # 1MB chunks
   845→        pos = start
   846→
   847→        while pos < end:
   848→            chunk_end = min(pos + chunk_size + footer_len, end + footer_len)
   849→            chunk = image[pos:chunk_end]
   850→
   851→            idx = chunk.find(footer)
   852→            if idx != -1:
   853→                return pos + idx
   854→
   855→            pos += chunk_size
   856→
   857→        return None
   858→
   859→    def _find_footer_last(self, image: bytes, start: int, end: int,
   860→                          footer: bytes) -> Optional[int]:
   861→        """
   862→        Find the LAST occurrence of footer bytes in the image.
   863→
   864→        Used for JPEG delimitation where the body may contain multiple
   865→        FFD9 sequences (EXIF thumbnails, random byte coincidences, or
   866→        embedded JPEG data). The real EOI marker is the last FFD9, not
   867→        the first one.
   868→
   869→        RC-002 fix: previous _find_footer() found the first FFD9, causing
   870→        severe truncation when the JPEG body contained spurious FFD9 bytes.
   871→        """
   872→        footer_len = len(footer)
   873→        if footer_len == 0:
   874→            return None
   875→
   876→        last_found = None
   877→        chunk_size = 1024 * 1024  # 1MB chunks
   878→        pos = start
   879→
   880→        while pos < end:
   881→            chunk_end = min(pos + chunk_size + footer_len, end + footer_len)
   882→            chunk = image[pos:chunk_end]
   883→
   884→            # Find ALL occurrences in this chunk
   885→            search_pos = 0
   886→            while search_pos < len(chunk):
   887→                idx = chunk.find(footer, search_pos)
   888→                if idx == -1:
   889→                    break
   890→                last_found = pos + idx
   891→                search_pos = idx + 1
   892→
   893→            pos += chunk_size
   894→
   895→        return last_found

   897→    def _carve_mp4(self, image: bytes, start_offset: int,
   898→                    image_len: int, sig: FileSignature) -> Optional[bytes]:
   899→        """
   900→        Carve an MP4 file by parsing the ftyp box to determine size.
   901→
   902→        MP4 structure: a series of boxes (atoms), each with:
   903→          [4 bytes size][4 bytes type][data...]
   904→        The first box is typically 'ftyp'.
   905→        """
   906→        # The 'ftyp' is at offset 4 within the first box
   907→        # So the first box starts at start_offset - 4
   908→        box_start = start_offset - 4
   909→
   910→        if box_start < 0 or box_start + 8 > image_len:
   911→            # Can't parse box header — use max-size heuristic
   912→            remaining = min(sig.max_size, image_len - start_offset)
   913→            return image[start_offset:start_offset + remaining]
   914→
   915→        # Read first box size
   916→        try:
   917→            box_size = struct.unpack('>I', image[box_start:box_start + 4])[0]
   918→        except:
   919→            remaining = min(sig.max_size, image_len - start_offset)
   920→            return image[start_offset:start_offset + remaining]
   921→
   922→        if box_size == 0:
   923→            # Box extends to end of file
   924→            return image[box_start:image_len]
   925→        elif box_size == 1:
   926→            # 64-bit extended size
   927→            if box_start + 16 > image_len:
   928→                remaining = min(sig.max_size, image_len - start_offset)
   929→                return image[start_offset:start_offset + remaining]
   930→            box_size = struct.unpack('>Q', image[box_start + 8:box_start + 16])[0]
   931→
   932→        # Sanity check
   933→        if box_size > sig.max_size or box_size < 8:
   934→            remaining = min(sig.max_size, image_len - start_offset)
   935→            return image[start_offset:start_offset + remaining]
   936→
   937→        end = box_start + box_size
   938→        if end > image_len:
   939→            end = image_len
   940→
   941→        return image[box_start:end]

   943→    def _resolve_zip_docx(self, carved_files: List[Dict]) -> List[Dict]:
   944→        """
   945→        Resolve ZIP/DOCX/XLSX ambiguity.
   946→
   947→        All three share the PK\x03\x04 header. We try to distinguish them
   948→        by looking for internal structure markers:
   949→          - DOCX contains "word/" or "word/document.xml"
   950→          - XLSX contains "xl/" or "xl/workbook.xml"
   951→          - ZIP contains other content
   952→
   953→        If we find multiple carves at the same offset (ZIP/DOCX/XLSX),
   954→        we keep the one that matches best.
   955→        """
   956→        # Group by start offset
   957→        by_offset: Dict[int, List[Dict]] = {}
   958→        for cf in carved_files:
   959→            offset = cf["start_offset"]
   960→            if offset not in by_offset:
   961→                by_offset[offset] = []
   962→            by_offset[offset].append(cf)
   963→
   964→        resolved = []
   965→        for offset, files in by_offset.items():
   966→            if len(files) == 1:
   967→                resolved.append(files[0])
   968→                continue
   969→
   970→            # Multiple carves at same offset — likely ZIP/DOCX/XLSX overlap
   971→            # Check internal content for format-specific markers
   972→            data = files[0]["data"]
   973→
   974→            has_docx_markers = (
   975→                b'word/' in data or
   976→                b'word/document.xml' in data or
   977→                b'Content_Types.xml' in data
   978→            )
   979→
   980→            has_xlsx_markers = (
   981→                b'xl/' in data or
   982→                b'xl/workbook.xml' in data or
   983→                b'xl/worksheets/' in data
   984→            )
   985→
   986→            if has_xlsx_markers:
   987→                # Keep the XLSX version
   988→                for f in files:
   989→                    if f["format_name"] == "XLSX":
   990→                        resolved.append(f)
   991→                        break
   992→                else:
   993→                    # No XLSX entry found — convert the first ZIP to XLSX
   994→                    files[0]["extension"] = ".xlsx"
   995→                    files[0]["format_name"] = "XLSX"
   996→                    resolved.append(files[0])
   997→            elif has_docx_markers:
   998→                # Keep the DOCX version
   999→                for f in files:
  1000→                    if f["format_name"] == "DOCX":
  1001→                        resolved.append(f)
  1002→                        break
  1003→                else:
  1004→                    # No DOCX entry found — convert the ZIP to DOCX
  1005→                    files[0]["extension"] = ".docx"
  1006→                    files[0]["format_name"] = "DOCX"
  1007→                    resolved.append(files[0])
  1008→            else:
  1009→                # Keep the ZIP version
  1010→                for f in files:
  1011→                    if f["format_name"] == "ZIP":
  1012→                        resolved.append(f)
  1013→                        break
  1014→                else:
  1015→                    resolved.append(files[0])
  1016→
  1017→        return resolved

  1019→    def _deduplicate_carves(self, carved_files: List[Dict]) -> List[Dict]:
  1020→        """
  1021→        Remove duplicate carves that overlap significantly.
  1022→
  1023→        If two carves overlap by >50% of the smaller one, keep the larger one.
  1024→        """
  1025→        if len(carved_files) <= 1:
  1026→            return carved_files
  1027→
  1028→        # Sort by start offset
  1029→        sorted_files = sorted(carved_files, key=lambda f: f["start_offset"])
  1030→
  1031→        deduped = []
  1032→        for cf in sorted_files:
  1033→            is_duplicate = False
  1034→
  1035→            for existing in deduped:
  1036→                # Check overlap
  1037→                overlap_start = max(cf["start_offset"], existing["start_offset"])
  1038→                overlap_end = min(
  1039→                    cf["start_offset"] + cf["size"],
  1040→                    existing["start_offset"] + existing["size"]
  1041→                )
  1042→
  1043→                if overlap_start < overlap_end:
  1044→                    overlap_size = overlap_end - overlap_start
  1045→                    smaller_size = min(cf["size"], existing["size"])
  1046→
  1047→                    if overlap_size > smaller_size * 0.5:
  1048→                        # Significant overlap — keep the larger one
  1049→                        if cf["size"] > existing["size"]:
  1050→                            deduped.remove(existing)
  1051→                            deduped.append(cf)
  1052→                        is_duplicate = True
  1053→                        break
  1054→
  1055→            if not is_duplicate:
  1056→                deduped.append(cf)
  1057→
  1058→        return deduped
