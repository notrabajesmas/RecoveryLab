    57→def add_styled_table(doc, headers, rows, col_widths=None):
    58→    table = doc.add_table(rows=1 + len(rows), cols=len(headers))
    59→    table.alignment = WD_TABLE_ALIGNMENT.CENTER
    60→    table.style = 'Table Grid'
    61→    for i, header in enumerate(headers):
    62→        cell = table.rows[0].cells[i]
    63→        cell.text = ""
    64→        p = cell.paragraphs[0]
    65→        run = p.add_run(header)
    66→        run.bold = True
    67→        run.font.size = Pt(10)
    68→        run.font.color.rgb = c(P["white"])
    69→        run.font.name = "Calibri"
    70→        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    71→        set_cell_shading(cell, P["primary"])
    72→    for row_idx, row_data in enumerate(rows):
    73→        for col_idx, cell_text in enumerate(row_data):
    74→            cell = table.rows[row_idx + 1].cells[col_idx]
    75→            cell.text = ""
    76→            p = cell.paragraphs[0]
    77→            run = p.add_run(str(cell_text))
    78→            run.font.size = Pt(9.5)
    79→            run.font.name = "Calibri"
    80→            run.font.color.rgb = c(P["body"])
    81→            if row_idx % 2 == 1:
    82→                set_cell_shading(cell, P["surface"])
    83→    if col_widths:
    84→        for row in table.rows:
    85→            for idx, width in enumerate(col_widths):
    86→                if idx < len(row.cells):
    87→                    row.cells[idx].width = Cm(width)
    88→    return table
