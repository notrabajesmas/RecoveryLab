   128→def compute_result_hash(metrics_dict: Dict) -> str:
   129→    hash_fields = {
   130→        "files_recovered": metrics_dict.get("files_recovered"),
   131→        "files_correct_checksum": metrics_dict.get("files_correct_checksum"),
   132→        "read_count": metrics_dict.get("read_count"),
   133→        "recovery_rate": metrics_dict.get("recovery_rate"),
   134→        "rvs": round(metrics_dict.get("rvs", 0), 6),
   135→        "weighted_functional_score": round(metrics_dict.get("weighted_functional_score", 0), 6),
   136→    }
   137→    hash_str = json.dumps(hash_fields, sort_keys=True)
   138→    return hashlib.sha256(hash_str.encode()).hexdigest()[:16]

   141→def apply_corruption(image: bytes, manifest: Dict, corruption_type: str,
   142→                     corruption_pct: float) -> Tuple[bytes, Dict]:
   143→    """
   144→    Apply simulated corruption to the NTFS image.
   145→
   146→    For MFT_PARTIAL_DELETE: Zero out a percentage of MFT entries.
   147→    This simulates the most common real-world scenario where MFT is
   148→    partially damaged.
   149→    """
   150→    if corruption_type == "NONE":
   151→        return image, manifest
   152→
   153→    # Get MFT info from manifest
   154→    mft_info = manifest.get("mft", {})
   155→    mft_start_cluster = mft_info.get("start_cluster", 0)
   156→    cluster_size = manifest.get("cluster_size", 4096)
   157→    mft_record_size = 1024
   158→
   159→    # Calculate MFT position in image
   160→    mft_offset = mft_start_cluster * cluster_size
   161→
   162→    # Get number of user files (non-system)
   163→    user_files = [f for f in manifest.get("files", []) if not f.get("is_directory", False)]
   164→    total_user_files = len(user_files)
   165→
   166→    # Calculate how many MFT entries to corrupt
   167→    # System files are records 0-11, user files start at 12
   168→    n_to_corrupt = int(total_user_files * corruption_pct)
   169→
   170→    # Create corrupted image
   171→    corrupted = bytearray(image)
   172→
   173→    # Zero out MFT entries (starting from user files = record 12)
   174→    mft_start_record = 12
   175→    for i in range(n_to_corrupt):
   176→        record_num = mft_start_record + i
   177→        record_offset = mft_offset + (record_num * mft_record_size)
   178→
   179→        # Don't corrupt beyond image bounds
   180→        if record_offset + mft_record_size > len(corrupted):
   181→            break
   182→
   183→        # Zero out the MFT record (simulate deletion)
   184→        for j in range(mft_record_size):
   185→            corrupted[record_offset + j] = 0
   186→
   187→    # Update manifest to reflect corruption
   188→    corrupted_manifest = dict(manifest)
   189→    corrupted_manifest["corruption"] = {
   190→        "type": corruption_type,
   191→        "percentage": corruption_pct,
   192→        "mft_entries_corrupted": n_to_corrupt,
   193→    }
   194→
   195→    return bytes(corrupted), corrupted_manifest

   198→def run_single_execution(run_index: int, image: bytes, manifest: Dict,
   199→                         motor_name: str = "MFT-First") -> Dict:
   200→    """Run a single execution and return all measurements."""
   201→    if motor_name == "MFT-First":
   202→        motor = MotorBMFTFirst()
   203→    elif motor_name == "Carving":
   204→        motor = MotorCarving()
   205→    else:
   206→        raise ValueError(f"Unknown motor: {motor_name}")
   207→
   208→    judge = RecoveryJudge(manifest)
   209→
   210→    t_start = time.perf_counter()
   211→    result = motor.recover(image, manifest, read_budget=0)
   212→    t_end = time.perf_counter()
   213→    runtime_ms = (t_end - t_start) * 1000.0
   214→
   215→    judge_input = [{
   216→        "name": f.name,
   217→        "sha256": f.sha256,
   218→        "size": f.size,
   219→        "is_directory": f.is_directory,
   220→        "data": f.data,
   221→    } for f in result.recovered_files]
   222→
   223→    metrics = judge.judge(
   224→        recovered_files=judge_input,
   225→        read_count=result.read_count,
   226→        sectors_wasted=result.sectors_wasted,
   227→        time_to_first_file=result.time_to_first_file,
   228→        mft_entries_parsed=result.mft_entries_parsed,
   229→    )
   230→
   231→    utility = compute_overall_utility(metrics.rvs, metrics.weighted_functional_score)
   232→    metrics_dict = metrics.to_dict()
   233→
   234→    return {
   235→        "run": run_index,
   236→        "motor": motor_name,
   237→        "timestamp": datetime.datetime.now().isoformat(),
   238→        "overall_utility": utility["overall_utility"],
   239→        "rvs": metrics.rvs,
   240→        "fqs": metrics.weighted_functional_score,
   241→        "recovery_rate": metrics.recovery_rate(),
   242→        "read_count": metrics.read_count,
   243→        "runtime_ms": runtime_ms,
   244→        "result_hash": compute_result_hash(metrics_dict),
   245→        "files_recovered": metrics_dict.get("files_recovered", 0),
   246→        "files_correct_checksum": metrics_dict.get("files_correct_checksum", 0),
   247→        "files_missing": metrics_dict.get("files_missing", 0),
   248→        "integrity_score": metrics_dict.get("integrity_score", 0.0),
   249→        "diagnostic": utility.get("diagnostic", ""),
   250→    }

   395→def generate_test_dataset_package(image: bytes, manifest: Dict,
   396→                                  corrupted_images: Dict, commit: str):
   397→    """Generate the standardized test dataset package for external tools."""
   398→    pkg_dir = OUTPUT_DIR / "test_dataset_package"
   399→    pkg_dir.mkdir(parents=True, exist_ok=True)
   400→
   401→    # Save healthy dataset
   402→    with open(pkg_dir / "healthy_10mb.img", 'wb') as f:
   403→        f.write(image)
   404→    save_manifest(manifest, pkg_dir / "healthy_10mb_manifest.json")
   405→
   406→    # Save corrupted variants
   407→    for name, (corr_image, corr_manifest) in corrupted_images.items():
   408→        with open(pkg_dir / f"{name}.img", 'wb') as f:
   409→            f.write(corr_image)
   410→        save_manifest(corr_manifest, pkg_dir / f"{name}_manifest.json")
   411→
   412→    # Generate README for external tools
   413→    readme = f"""# EXP-0005 — External Tool Test Dataset Package
   414→==============================================
   415→Generated: {datetime.datetime.now().strftime('%Y-%m-%d %H:%M')}
   416→Commit: {commit}
   417→Seed: {SEED}
   418→
   419→## Files
   420→- healthy_10mb.img / _manifest.json — Healthy NTFS image (0% corruption)
   421→- mft20_10mb.img / _manifest.json — 20% MFT entries zeroed
   422→- mft60_10mb.img / _manifest.json — 60% MFT entries zeroed
   423→
   424→## How to Test an External Tool
   425→1. Run the tool on each .img file
   426→2. Record:
   427→   - Total files recovered
   428→   - Files with correct content (compare with manifest SHA-256)
   429→   - Total runtime
   430→   - Read count (if available)
   431→3. Compute OU = RVS × FQS using the manifest values
   432→4. Add results to the comparison table in external_validation_report.md
   433→
   434→## Expected Results (RecoveryLab)
   435→See external_validation_summary.json for RecoveryLab's results on these datasets.
   436→
   437→## Important
   438→- The manifest.json contains the GROUND TRUTH (what's in the image)
   439→- Compare the tool's output against the manifest to compute recovery metrics
   440→- Use the same Judge API (or equivalent) for fair comparison

   442→    with open(pkg_dir / "README.md", 'w') as f:
   443→        f.write(readme)

   445→    # Save metadata
   446→    metadata = {
   447→        "experiment_id": EXPERIMENT_ID,
   448→        "commit": commit,
   449→        "seed": SEED,
   450→        "datasets": [c["name"] for c in TEST_CONFIGS],

   453→    with open(pkg_dir / "package_metadata.json", 'w') as f:
   454→        json.dump(metadata, f, indent=2, default=str)
   455→
   456→    return pkg_dir
