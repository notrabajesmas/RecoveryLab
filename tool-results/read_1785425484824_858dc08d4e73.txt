     1→"""
     2→RecoveryLab — Hypothesis Registry
     3→====================================
     4→Formal tracking of hypotheses, evidence, and status.
     5→
     6→The goal is NOT "add features" — it's "close hypotheses with evidence."
     7→
     8→Each hypothesis has:
     9→  - ID: Unique identifier (H1.1, H1.2, H1.3, ...)
    10→  - Statement: What we're testing
    11→  - Status: pending | in_evaluation | supported | refuted | refined
    12→  - Evidence: What experiments support or refute this
    13→  - Confidence: How confident we are (0.0-1.0)
    14→  - Dependencies: What hypotheses must be resolved first
    15→  - Open questions: What we still don't know
    16→
    17→The registry is append-only. Once a hypothesis is supported or refuted,
    18→it stays that way. If it's refined, the refinement becomes a new hypothesis.
    19→
    20→This is the single source of truth for the project's scientific direction.
    21→"""
    22→
    23→import json
    24→from datetime import datetime, timezone
    25→from pathlib import Path
    26→from typing import List, Dict, Optional
    27→from dataclasses import dataclass, field
    28→from enum import Enum
    29→
    30→
    31→class HypothesisStatus(Enum):
    32→    PENDING = "pending"
    33→    IN_EVALUATION = "in_evaluation"
    34→    SUPPORTED = "supported"
    35→    REFUTED = "refuted"
    36→    REFINED = "refined"       # Original was too broad; split into sub-hypotheses
    37→    INCONCLUSIVE = "inconclusive"
    38→
    39→
    40→class EvidenceType(Enum):
    41→    SIMULATION = "simulation"       # From RecoveryLab experiments
    42→    EXTERNAL_TOOL = "external_tool" # From TestDisk, PhotoRec, etc.
    43→    REAL_DISK = "real_disk"         # From actual hardware
    44→    LITERATURE = "literature"       # From published research
    45→    FORMAL = "formal"               # From mathematical proof
    46→
    47→
    48→@dataclass
    49→class Evidence:
    50→    """A piece of evidence supporting or refuting a hypothesis."""
    51→    timestamp: str
    52→    type: EvidenceType
    53→    supports: bool                  # True = supports, False = refutes
    54→    description: str
    55→    experiment_id: str = ""         # Link to experiment results
    56→    strength: str = "moderate"      # weak | moderate | strong
    57→    details: Dict = field(default_factory=dict)
    58→
    59→
    60→@dataclass
    61→class Hypothesis:
    62→    """A single hypothesis with full tracking."""
    63→    id: str
    64→    statement: str
    65→    status: HypothesisStatus = HypothesisStatus.PENDING
    66→    confidence: float = 0.0
    67→    evidence: List[Evidence] = field(default_factory=list)
    68→    dependencies: List[str] = field(default_factory=list)
    69→    open_questions: List[str] = field(default_factory=list)
    70→    refinement_of: Optional[str] = None   # If this refines a previous hypothesis
    71→    created_at: str = ""
    72→    updated_at: str = ""
    73→
    74→    def add_evidence(self, evidence: Evidence):
    75→        """Add evidence and update status/confidence."""
    76→        self.evidence.append(evidence)
    77→        self.updated_at = datetime.now(timezone.utc).isoformat()
    78→        self._recompute_status()
    79→
    80→    def _recompute_status(self):
    81→        """Recompute status based on accumulated evidence."""
    82→        if not self.evidence:
    83→            self.status = HypothesisStatus.PENDING
    84→            return
    85→
    86→        supports = [e for e in self.evidence if e.supports]
    87→        refutes = [e for e in self.evidence if not e.supports]
    88→
    89→        support_strength = sum(
    90→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    91→            for e in supports
    92→        )
    93→        refute_strength = sum(
    94→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    95→            for e in refutes
    96→        )
    97→
    98→        total = support_strength + refute_strength
    99→        if total == 0:
   100→            self.confidence = 0.5
   101→            self.status = HypothesisStatus.INCONCLUSIVE
   102→            return
   103→
   104→        self.confidence = support_strength / total
   105→
   106→        if self.confidence >= 0.8:
   107→            self.status = HypothesisStatus.SUPPORTED
   108→        elif self.confidence <= 0.2:
   109→            self.status = HypothesisStatus.REFUTED
   110→        elif len(self.evidence) < 3:
   111→            self.status = HypothesisStatus.IN_EVALUATION
   112→        else:
   113→            self.status = HypothesisStatus.INCONCLUSIVE
   114→
   115→    def to_dict(self) -> Dict:
   116→        return {
   117→            "id": self.id,
   118→            "statement": self.statement,
   119→            "status": self.status.value,
   120→            "confidence": round(self.confidence, 3),
   121→            "evidence": [
   122→                {
   123→                    "timestamp": e.timestamp,
   124→                    "type": e.type.value,
   125→                    "supports": e.supports,
   126→                    "description": e.description,
   127→                    "experiment_id": e.experiment_id,
   128→                    "strength": e.strength,
   129→                }
   130→                for e in self.evidence
   131→            ],
   132→            "dependencies": self.dependencies,
   133→            "open_questions": self.open_questions,
   134→            "refinement_of": self.refinement_of,
   135→            "created_at": self.created_at,
   136→            "updated_at": self.updated_at,
   137→        }

   599→    def register(self, hypothesis_id: str, statement: str,
   600→                 status: HypothesisStatus = HypothesisStatus.PENDING,
   601→                 dependencies: List[str] = None,
   602→                 open_questions: List[str] = None,
   603→                 refinement_of: str = None) -> Hypothesis:
   604→        """Register a new hypothesis."""
   605→        if hypothesis_id in self.hypotheses:
   606→            return self.hypotheses[hypothesis_id]
   607→
   608→        now = datetime.now(timezone.utc).isoformat()
   609→        h = Hypothesis(
   610→            id=hypothesis_id,
   611→            statement=statement,
   612→            status=status,
   613→            dependencies=dependencies or [],
   614→            open_questions=open_questions or [],
   615→            refinement_of=refinement_of,
   616→            created_at=now,
   617→            updated_at=now,
   618→        )
   619→        self.hypotheses[hypothesis_id] = h
   620→        return h

   622→    def add_evidence(self, hypothesis_id: str, evidence: Evidence):
   623→        """Add evidence to a hypothesis."""
   624→        if hypothesis_id not in self.hypotheses:
   625→            raise ValueError(f"Unknown hypothesis: {hypothesis_id}")
   626→        self.hypotheses[hypothesis_id].add_evidence(evidence)

   629→        """Get a hypothesis by ID."""

   632→    def all_hypotheses(self) -> List[Hypothesis]:
   633→        """Get all hypotheses sorted by ID."""
   634→        return sorted(self.hypotheses.values(), key=lambda h: h.id)

   636→    def summary_table(self) -> str:
   637→        """Generate a markdown summary table of all hypotheses."""
   638→        lines = [
   639→            "# RecoveryLab — Hypothesis Registry",
   640→            "",
   641→            "| ID | Hypothesis | Status | Confidence | Evidence | Open Questions |",
   642→            "|----|-----------|--------|------------|----------|----------------|",
   643→        ]
   644→        for h in self.all_hypotheses():
   645→            supports = sum(1 for e in h.evidence if e.supports)
   646→            refutes = sum(1 for e in h.evidence if not e.supports)
   647→            evidence_str = f"{supports}S / {refutes}R"
   648→            questions = len(h.open_questions)
   649→            lines.append(
   650→                f"| {h.id} | {h.statement[:60]}... | {h.status.value} | "
   651→                f"{h.confidence:.0%} | {evidence_str} | {questions} |"
   652→            )
   653→        return "\n".join(lines)

   655→    def detailed_report(self) -> str:
   656→        """Generate a detailed report of all hypotheses."""
   657→        lines = ["# RecoveryLab — Detailed Hypothesis Report", ""]
   658→
   659→        for h in self.all_hypotheses():
   660→            lines.append(f"## {h.id}: {h.statement}")
   661→            lines.append(f"**Status:** {h.status.value}")
   662→            lines.append(f"**Confidence:** {h.confidence:.0%}")
   663→            if h.refinement_of:
   664→                lines.append(f"**Refinement of:** {h.refinement_of}")
   665→            if h.dependencies:
   666→                lines.append(f"**Dependencies:** {', '.join(h.dependencies)}")
   667→            lines.append("")
   668→
   669→            if h.evidence:
   670→                lines.append("### Evidence")
   671→                for i, e in enumerate(h.evidence, 1):
   672→                    symbol = "✓" if e.supports else "✗"

   690→    def save(self, path: Path):
   691→        """Save the registry to a JSON file."""
   692→        data = {
   693→            "generated_at": datetime.now(timezone.utc).isoformat(),
   694→            "hypotheses": {h_id: h.to_dict() for h_id, h in self.hypotheses.items()},
   695→        }
   696→        path.parent.mkdir(parents=True, exist_ok=True)
   697→        with open(path, 'w') as f:
   698→            json.dump(data, f, indent=2, ensure_ascii=False)

   700→    def load(self, path: Path):
   701→        """Load the registry from a JSON file."""
   702→        with open(path, 'r') as f:
   703→            data = json.load(f)
   704→
   705→        for h_id, h_data in data.get("hypotheses", {}).items():
   706→            if h_id in self.hypotheses:
   707→                # Update existing hypothesis with loaded evidence
   708→                h = self.hypotheses[h_id]
   709→                for e_data in h_data.get("evidence", []):
   710→                    # Check if we already have this evidence
   711→                    already_exists = any(
   712→                        e.description == e_data["description"]
   713→                        for e in h.evidence
   714→                    )
   715→                    if not already_exists:
   716→                        h.add_evidence(Evidence(
   717→                            timestamp=e_data["timestamp"],
   718→                            type=EvidenceType(e_data["type"]),
   719→                            supports=e_data["supports"],
   720→                            description=e_data["description"],
   721→                            experiment_id=e_data.get("experiment_id", ""),
   722→                            strength=e_data.get("strength", "moderate"),
   723→                        ))
