     1→     1→"""
     2→     2→RecoveryLab — Hypothesis Registry
     3→     3→====================================
     4→     4→Formal tracking of hypotheses, evidence, and status.
     5→     5→
     6→     6→The goal is NOT "add features" — it's "close hypotheses with evidence."
     7→     7→
     8→     8→Each hypothesis has:
     9→     9→  - ID: Unique identifier (H1.1, H1.2, H1.3, ...)
    10→    10→  - Statement: What we're testing
    11→    11→  - Status: pending | in_evaluation | supported | refuted | refined
    12→    12→  - Evidence: What experiments support or refute this
    13→    13→  - Confidence: How confident we are (0.0-1.0)
    14→    14→  - Dependencies: What hypotheses must be resolved first
    15→    15→  - Open questions: What we still don't know
    16→    16→
    17→    17→The registry is append-only. Once a hypothesis is supported or refuted,
    18→    18→it stays that way. If it's refined, the refinement becomes a new hypothesis.
    19→    19→
    20→    20→This is the single source of truth for the project's scientific direction.
    21→    21→"""
    22→    22→
    23→    23→import json
    24→    24→from datetime import datetime, timezone

    29→    29→
    30→    30→
    31→    31→class HypothesisStatus(Enum):
    32→    32→    PENDING = "pending"
    33→    33→    IN_EVALUATION = "in_evaluation"
    34→    34→    SUPPORTED = "supported"
    35→    35→    REFUTED = "refuted"
    36→    36→    REFINED = "refined"       # Original was too broad; split into sub-hypotheses
    37→    37→    INCONCLUSIVE = "inconclusive"
    38→    38→
    39→    39→
    40→    40→class EvidenceType(Enum):
    41→    41→    SIMULATION = "simulation"       # From RecoveryLab experiments
    42→    42→    EXTERNAL_TOOL = "external_tool" # From TestDisk, PhotoRec, etc.
    43→    43→    REAL_DISK = "real_disk"         # From actual hardware
    44→    44→    LITERATURE = "literature"       # From published research
    45→    45→    FORMAL = "formal"               # From mathematical proof
    46→    46→
    47→    47→
    48→    48→@dataclass
    49→    49→class Evidence:
    50→    50→    """A piece of evidence supporting or refuting a hypothesis."""
    51→    51→    timestamp: str
    52→    52→    type: EvidenceType
    53→    53→    supports: bool                  # True = supports, False = refutes
    54→    54→    description: str
    55→    55→    experiment_id: str = ""         # Link to experiment results
    56→    56→    strength: str = "moderate"      # weak | moderate | strong
    57→    57→    details: Dict = field(default_factory=dict)
    58→    58→
    59→    59→
    60→    60→@dataclass
    61→    61→class Hypothesis:
    62→    62→    """A single hypothesis with full tracking."""
    63→    63→    id: str
    64→    64→    statement: str
    65→    65→    status: HypothesisStatus = HypothesisStatus.PENDING
    66→    66→    confidence: float = 0.0
    67→    67→    evidence: List[Evidence] = field(default_factory=list)
    68→    68→    dependencies: List[str] = field(default_factory=list)
    69→    69→    open_questions: List[str] = field(default_factory=list)
    70→    70→    refinement_of: Optional[str] = None   # If this refines a previous hypothesis
    71→    71→    created_at: str = ""
    72→    72→    updated_at: str = ""
    73→    73→
    74→    74→    def add_evidence(self, evidence: Evidence):
    75→    75→        """Add evidence and update status/confidence."""
    76→    76→        self.evidence.append(evidence)
    77→    77→        self.updated_at = datetime.now(timezone.utc).isoformat()
    78→    78→        self._recompute_status()
    79→    79→
    80→    80→    def _recompute_status(self):
    81→    81→        """Recompute status based on accumulated evidence."""
    82→    82→        if not self.evidence:
    83→    83→            self.status = HypothesisStatus.PENDING
    84→    84→            return
    85→    85→
    86→    86→        supports = [e for e in self.evidence if e.supports]
    87→    87→        refutes = [e for e in self.evidence if not e.supports]
    88→    88→
    89→    89→        support_strength = sum(
    90→    90→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    91→    91→            for e in supports
    92→    92→        )
    93→    93→        refute_strength = sum(
    94→    94→            {"weak": 1, "moderate": 2, "strong": 3}[e.strength]
    95→    95→            for e in refutes
    96→    96→        )
    97→    97→
    98→    98→        total = support_strength + refute_strength
    99→    99→        if total == 0:
   100→   100→            self.confidence = 0.5
   101→   101→            self.status = HypothesisStatus.INCONCLUSIVE
   102→   102→            return
   103→   103→
   104→   104→        self.confidence = support_strength / total
   105→   105→
   106→   106→        if self.confidence >= 0.8:
   107→   107→            self.status = HypothesisStatus.SUPPORTED
   108→   108→        elif self.confidence <= 0.2:
   109→   109→            self.status = HypothesisStatus.REFUTED
   110→   110→        elif len(self.evidence) < 3:
   111→   111→            self.status = HypothesisStatus.IN_EVALUATION
   112→   112→        else:
   113→   113→            self.status = HypothesisStatus.INCONCLUSIVE
   114→   114→
