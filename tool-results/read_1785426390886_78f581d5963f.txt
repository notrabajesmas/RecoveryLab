   698→    def register(self, hypothesis_id: str, statement: str,
   699→                 status: HypothesisStatus = HypothesisStatus.PENDING,
   700→                 dependencies: List[str] = None,
   701→                 open_questions: List[str] = None,
   702→                 refinement_of: str = None) -> Hypothesis:
   703→        """Register a new hypothesis."""
   704→        if hypothesis_id in self.hypotheses:
   705→            return self.hypotheses[hypothesis_id]
   706→
   707→        now = datetime.now(timezone.utc).isoformat()
   708→        h = Hypothesis(
   709→            id=hypothesis_id,
   710→            statement=statement,
   711→            status=status,
   712→            dependencies=dependencies or [],
   713→            open_questions=open_questions or [],
   714→            refinement_of=refinement_of,
   715→            created_at=now,
   716→            updated_at=now,
   717→        )
   718→        self.hypotheses[hypothesis_id] = h
   719→        return h

   721→    def add_evidence(self, hypothesis_id: str, evidence: Evidence):
   722→        """Add evidence to a hypothesis."""
   723→        if hypothesis_id not in self.hypotheses:
   724→            raise ValueError(f"Unknown hypothesis: {hypothesis_id}")
   725→        self.hypotheses[hypothesis_id].add_evidence(evidence)

   727→    def get(self, hypothesis_id: str) -> Optional[Hypothesis]:
   728→        """Get a hypothesis by ID."""
   729→        return self.hypotheses.get(hypothesis_id)

   731→    def all_hypotheses(self) -> List[Hypothesis]:
   732→        """Get all hypotheses sorted by ID."""
   733→        return sorted(self.hypotheses.values(), key=lambda h: h.id)

   735→    def summary_table(self) -> str:
   736→        """Generate a markdown summary table of all hypotheses."""
   737→        lines = [
   738→            "# RecoveryLab — Hypothesis Registry",
   739→            "",
   740→            "| ID | Hypothesis | Status | Confidence | Evidence | Open Questions |",
   741→            "|----|-----------|--------|------------|----------|----------------|",
   742→        ]
   743→        for h in self.all_hypotheses():
   744→            supports = sum(1 for e in h.evidence if e.supports)
   745→            refutes = sum(1 for e in h.evidence if not e.supports)
   746→            evidence_str = f"{supports}S / {refutes}R"
   747→            questions = len(h.open_questions)
   748→            lines.append(
   749→                f"| {h.id} | {h.statement[:60]}... | {h.status.value} | "
   750→                f"{h.confidence:.0%} | {evidence_str} | {questions} |"
   751→            )
   752→        return "\n".join(lines)

   754→    def detailed_report(self) -> str:
   755→        """Generate a detailed report of all hypotheses."""
   756→        lines = ["# RecoveryLab — Detailed Hypothesis Report", ""]
   757→
   758→        for h in self.all_hypotheses():
   759→            lines.append(f"## {h.id}: {h.statement}")
   760→            lines.append(f"**Status:** {h.status.value}")
   761→            lines.append(f"**Confidence:** {h.confidence:.0%}")
   762→            if h.refinement_of:
   763→                lines.append(f"**Refinement of:** {h.refinement_of}")
   764→            if h.dependencies:
   765→                lines.append(f"**Dependencies:** {', '.join(h.dependencies)}")
   766→            lines.append("")
   767→
   768→            if h.evidence:
   769→                lines.append("### Evidence")
   770→                for i, e in enumerate(h.evidence, 1):
   771→                    symbol = "✓" if e.supports else "✗"
   772→                    lines.append(
   773→                        f"{i}. {symbol} [{e.type.value}] {e.description} "
   774→                        f"(strength: {e.strength})"
   775→                    )
   776→                lines.append("")
   777→
   778→            if h.open_questions:
   779→                lines.append("### Open Questions")
   780→                for q in h.open_questions:
   781→                    lines.append(f"- {q}")
   782→                lines.append("")
   783→
   784→            lines.append("---")
   785→            lines.append("")

   787→        return "\n".join(lines)

   789→    def save(self, path: Path):
   790→        """Save the registry to a JSON file."""
   791→        data = {
   792→            "generated_at": datetime.now(timezone.utc).isoformat(),
   793→            "hypotheses": {h_id: h.to_dict() for h_id, h in self.hypotheses.items()},
   794→        }
   795→        path.parent.mkdir(parents=True, exist_ok=True)
   796→        with open(path, 'w') as f:
   797→            json.dump(data, f, indent=2, ensure_ascii=False)

   799→    def load(self, path: Path):
   800→        """Load the registry from a JSON file."""
   801→        with open(path, 'r') as f:
   802→            data = json.load(f)
   803→
   804→        for h_id, h_data in data.get("hypotheses", {}).items():
   805→            if h_id in self.hypotheses:
   806→                # Update existing hypothesis with loaded evidence
   807→                h = self.hypotheses[h_id]
   808→                for e_data in h_data.get("evidence", []):
   809→                    # Check if we already have this evidence
   810→                    already_exists = any(
   811→                        e.description == e_data["description"]
   812→                        for e in h.evidence
   813→                    )
   814→                    if not already_exists:
   815→                        h.add_evidence(Evidence(
   816→                            timestamp=e_data["timestamp"],
   817→                            type=EvidenceType(e_data["type"]),
   818→                            supports=e_data["supports"],
   819→                            description=e_data["description"],
   820→                            experiment_id=e_data.get("experiment_id", ""),
   821→                            strength=e_data.get("strength", "moderate"),
   822→                        ))

   824→    def next_steps(self) -> List[str]:
   825→        """What experiments should we run next to close open hypotheses?"""
   826→        steps = []
   827→
   828→        # Check what's most needed
   829→        for h in self.all_hypotheses():
   830→            if h.status == HypothesisStatus.PENDING:
   831→                steps.append(
   832→                    f"[{h.id}] PENDING — needs first experiment: {h.open_questions[0] if h.open_questions else 'design experiment'}"
   833→                )
   834→            elif h.status == HypothesisStatus.IN_EVALUATION:
   835→                steps.append(
   836→                    f"[{h.id}] IN EVALUATION — needs more evidence: {h.open_questions[0] if h.open_questions else 'gather more data'}"
   837→                )
   838→            elif h.status == HypothesisStatus.INCONCLUSIVE:
   839→                steps.append(
   840→                    f"[{h.id}] INCONCLUSIVE — conflicting evidence, need decisive experiment"
   841→                )
   842→
   843→        return steps
