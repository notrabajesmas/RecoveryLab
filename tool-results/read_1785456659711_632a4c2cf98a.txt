    77→def run_motor_on_dataset(motor_name: str, image: bytes, manifest: Dict) -> Dict:
    78→    """Run a single motor on a dataset and return raw results + diagnostics."""
    79→    if motor_name == "MFT-First":
    80→        motor = MotorBMFTFirst()
    81→    elif motor_name == "Carving":
    82→        motor = MotorCarving()
    83→    else:
    84→        raise ValueError(f"Unknown motor: {motor_name}")
    85→
    86→    judge = RecoveryJudge(manifest)
    87→
    88→    t_start = time.perf_counter()
    89→    result = motor.recover(image, manifest, read_budget=0)
    90→    t_end = time.perf_counter()
    91→    runtime_ms = (t_end - t_start) * 1000.0
    92→
    93→    carving_stats = getattr(result, 'carving_stats', {})
    94→
    95→    judge_input = [{
    96→        "name": f.name, "sha256": f.sha256, "size": f.size,
    97→        "is_directory": f.is_directory, "data": f.data,
    98→    } for f in result.recovered_files]
    99→
   100→    metrics = judge.judge(
   101→        recovered_files=judge_input,
   102→        read_count=result.read_count,
   103→        sectors_wasted=result.sectors_wasted,
   104→        time_to_first_file=result.time_to_first_file,
   105→        mft_entries_parsed=result.mft_entries_parsed,
   106→    )
   107→
   108→    utility = compute_overall_utility(metrics.rvs, metrics.weighted_functional_score)
   109→    metrics_dict = metrics.to_dict()
   110→
   111→    # ── Per-file diagnostics ──
   112→    per_file_details = []
   113→    for rf in result.recovered_files:
   114→        gt_match = None
   115→        match_method = "none"
   116→
   117→        gt_file_by_name = judge.ground_truth["files_by_name"].get(rf.name)
   118→        if gt_file_by_name:
   119→            gt_match = gt_file_by_name
   120→            match_method = "name"
   121→        else:
   122→            gt_file_by_sha = judge.ground_truth["files_by_sha"].get(rf.sha256)
   123→            if gt_file_by_sha:
   124→                gt_match = gt_file_by_sha
   125→                match_method = "sha256"
   126→
   127→        # Determine failure reason
   128→        failure_reason = ""
   129→        size_diff = 0
   130→        truncation_fix = False
   131→
   132→        if match_method == "none":
   133→            # Check if truncation would match
   134→            for gt_sha, gt_f in judge.ground_truth["files_by_sha"].items():
   135→                if rf.size > gt_f.get("size", 0) and rf.data is not None:
   136→                    truncated = rf.data[:gt_f["size"]]
   137→                    if hashlib.sha256(truncated).hexdigest() == gt_sha:
   138→                        size_diff = rf.size - gt_f["size"]
   139→                        failure_reason = f"Cluster padding: {size_diff} extra bytes after EOF. Truncation fixes SHA-256."
   140→                        truncation_fix = True
   141→                        break
   142→                elif rf.size < gt_f.get("size", 0) and rf.data is not None:
   143→                    # Check if adding bytes from image would match
   144→                    # (This is the PDF case: missing \n after %%EOF)
   145→                    gt_clusters = gt_f.get("clusters", [])
   146→                    if gt_clusters:
   147→                        gt_start = gt_clusters[0] * 4096
   148→                        gt_data = image[gt_start:gt_start + gt_f["size"]]
   149→                        if rf.data == gt_data[:rf.size]:
   150→                            size_diff = rf.size - gt_f["size"]
   151→                            missing_byte = gt_data[rf.size:rf.size+1]
   152→                            failure_reason = f"Missing {abs(size_diff)} byte(s) at end: {missing_byte.hex()} = {missing_byte!r}. Adding it fixes SHA-256."
   153→                            truncation_fix = True
   154→                            break
   155→
   156→            if not truncation_fix:
   157→                if rf.name.startswith("carved_"):
   158→                    failure_reason = "SHA-256 not in ground truth. Data differs from original."
   159→                else:
   160→                    failure_reason = "Name not in ground truth and SHA-256 not in ground truth."
   161→
   162→        per_file_details.append({
   163→            "recovered_name": rf.name,
   164→            "recovered_sha256": rf.sha256,
   165→            "recovered_size": rf.size,
   166→            "gt_match": gt_match.get("name", "NONE") if gt_match else "NONE",
   167→            "match_method": match_method,
   168→            "size_diff": size_diff,
   169→            "truncation_fixes_sha256": truncation_fix,
   170→            "failure_reason": failure_reason,
   171→        })
   172→
   173→    return {
   174→        "motor": motor_name,
   175→        "runtime_ms": runtime_ms,
   176→        "overall_utility": utility["overall_utility"],
   177→        "rvs": metrics.rvs,
   178→        "fqs": metrics.weighted_functional_score,
   179→        "recovery_rate": metrics.recovery_rate(),
   180→        "files_recovered": metrics_dict.get("files_recovered", 0),
   181→        "files_correct_checksum": metrics_dict.get("files_correct_checksum", 0),
   182→        "files_missing": metrics_dict.get("files_missing", 0),
   183→        "false_positives": metrics.false_positives,
   184→        "read_count": metrics.read_count,
   185→        "mft_entries_parsed": metrics.mft_entries_parsed,
   186→        "carving_stats": carving_stats,
   187→        "per_file_details": per_file_details,
   188→        "result_hash": hashlib.sha256(
   189→            json.dumps({
   190→                "files_recovered": metrics_dict.get("files_recovered"),
   191→                "rvs": round(metrics.rvs, 6),
   192→                "fqs": round(metrics.weighted_functional_score, 6),
   193→            }, sort_keys=True).encode()
   194→        ).hexdigest()[:16],
   195→    }

   198→def diagnose_carving_zero(format_results: Dict) -> Dict:
   199→    """Diagnose the origin of the carving zero based on all format results."""
   200→    diagnosis = {
   201→        "question": "¿El cero proviene del algoritmo o del banco de pruebas?",
   202→        "format_analysis": {},
   203→        "zero_origin": "UNDETERMINED",
   204→        "zero_origin_explanation": "",
   205→        "hypothesis_ranking": [],
   206→        "root_causes": [],
   207→    }
   208→
   209→    formats_with_nonzero_carving = []
   210→    formats_with_zero_carving = []
   211→    formats_with_truncation_fix = []
   212→
   213→    for fmt_id, results in format_results.items():
   214→        cr = results["Carving"]
   215→        mr = results["MFT-First"]
   216→        carving_ou = cr["overall_utility"]
   217→        sigs_found = cr["carving_stats"].get("signatures_found", {})
   218→        files_carved = cr["carving_stats"].get("files_carved", 0)
   219→        files_recovered = cr["files_recovered"]
   220→        files_correct = cr["files_correct_checksum"]
   221→        false_positives = cr["false_positives"]
   222→        matched = sum(1 for d in cr["per_file_details"] if d["match_method"] != "none")
   223→        truncation_fixable = sum(1 for d in cr["per_file_details"] if d["truncation_fixes_sha256"])
   224→
   225→        format_diag = {
   226→            "carving_ou": carving_ou,
   227→            "mft_ou": mr["overall_utility"],
   228→            "signatures_found": sigs_found,
   229→            "files_carved": files_carved,
   230→            "files_recovered_by_judge": files_recovered,
   231→            "files_correct_checksum": files_correct,
   232→            "false_positives": false_positives,
   233→            "matched_to_gt": matched,
   234→            "unmatched_to_gt": files_carved - matched,
   235→            "truncation_fixable": truncation_fixable,
   236→            "per_file_details": cr["per_file_details"],
   237→        }
   238→
   239→        diagnosis["format_analysis"][fmt_id] = format_diag
   240→
   241→        if carving_ou > 0.0:
   242→            formats_with_nonzero_carving.append(fmt_id)
   243→        else:
   244→            formats_with_zero_carving.append(fmt_id)
   245→
   246→        if truncation_fixable > 0:
   247→            formats_with_truncation_fix.append(fmt_id)
   248→
   249→    # ── Determine the origin of the zero ──
   250→    # Key findings from the data:
   251→    # - ZIP/DOCX: OU = 1.0 (carving works perfectly)
   252→    # - PNG: OU > 0 (carving works well, minor issues)
   253→    # - PDF: OU = 0.0 but carved files are only 1 byte short (missing \n after %%EOF)
   254→    # - JPEG: OU = 0.0, only 3/15 carved, carved files are too short
   255→
   256→    root_causes = []
   257→
   258→    # Check for PDF footer mismatch
   259→    pdf_diag = diagnosis["format_analysis"].get("C", {})
   260→    if pdf_diag.get("truncation_fixable", 0) > 0:
   261→        root_causes.append({
   262→            "id": "RC-001",
   263→            "format": "PDF",
   264→            "cause": "Footer mismatch: carving motor uses %%EOF (5 bytes) but file generator produces %%EOF\\n (6 bytes)",
   265→            "impact": "ALL carved PDF files are 1 byte short. SHA-256 doesn't match.",
   266→            "fix": "Change PDF signature footer from %%EOF to %%EOF\\n in motor_carving.py",
   267→            "severity": "HIGH — causes 100% PDF carving failure",
   268→        })
   269→
   270→    # Check for JPEG carving issues
   271→    jpeg_diag = diagnosis["format_analysis"].get("A", {})
   272→    if jpeg_diag.get("files_carved", 0) < jpeg_diag.get("signatures_found", {}).get("JPEG", 0):
   273→        root_causes.append({
   274→            "id": "RC-002",
   275→            "format": "JPEG",
   276→            "cause": "Deduplication/overlap removal: carving motor finds 15 JPEG signatures but only carves 3 files",
   277→            "impact": "Most JPEG files are not carved. The 3 carved files are too short (missing millions of bytes).",
   278→            "fix": "Investigate why _deduplicate_carves removes 12/15 JPEG carves. Likely: large files overlap with other signatures.",
   279→            "severity": "HIGH — causes near-total JPEG carving failure",
   280→        })
   281→
   282→    # Check for PNG minor issues
   283→    png_diag = diagnosis["format_analysis"].get("B", {})
   284→    if png_diag.get("false_positives", 0) > 0:
   285→        root_causes.append({
   286→            "id": "RC-003",
   287→            "format": "PNG",
   288→            "cause": "1 false positive: BMP signature detected within PNG data",
   289→            "impact": "Minor — 14/15 PNG files recovered correctly",
   290→            "fix": "Improve BMP signature detection to avoid false positives within PNG data",
   291→            "severity": "LOW — minor impact on PNG carving",
   292→        })
   293→
   294→    diagnosis["root_causes"] = root_causes
   295→
   296→    # Determine overall diagnosis
   297→    if len(formats_with_nonzero_carving) >= 2 and len(formats_with_zero_carving) >= 1:
   298→        diagnosis["zero_origin"] = "FORMAT_SPECIFIC_PARSER_ISSUES"
   299→        diagnosis["zero_origin_explanation"] = (
   300→            f"El carving funciona para {formats_with_nonzero_carving} pero falla para "
   301→            f"{formats_with_zero_carving}. Las causas raíz son específicas por formato: "
   302→            f"PDF tiene un bug de footer (1 byte), JPEG tiene un problema de deduplicación. "
   303→            f"El motor de carving en sí funciona — el scanner encuentra firmas correctamente. "
   304→            f"El problema está en la extracción (footer detection + deduplication), no en "
   305→            f"la detección (signature scanning)."
   306→        )
   307→    elif len(formats_with_nonzero_carving) == len(FORMAT_DATASETS):
   308→        diagnosis["zero_origin"] = "NO_ZERO_IN_SINGLE_FORMAT"
   309→        diagnosis["zero_origin_explanation"] = "El carving funciona para todos los formatos individuales."
   310→    else:
   311→        diagnosis["zero_origin"] = "MOTOR_LEVEL_ISSUE"
   312→        diagnosis["zero_origin_explanation"] = "El carving falla para todos los formatos."
   313→
   314→    # ── Rank hypotheses ──
   315→    hypothesis_ranking = [
   316→        {
   317→            "id": "H-CARVING-001",
   318→            "statement": "El parser de carving tiene un bug",
   319→            "probability": "HIGH",
   320→            "evidence": "PDF footer bug: %%EOF vs %%EOF\\n. JPEG deduplication bug: 12/15 files removed.",
   321→            "refined": "No es un bug general del parser, sino bugs específicos: footer de PDF y deduplicación de JPEG.",
   322→        },
   323→        {
   324→            "id": "H-CARVING-003",
   325→            "statement": "El generador produce archivos poco realistas",
   326→            "probability": "LOW",
   327→            "evidence": "ZIP/DOCX/PNG carving funciona perfectamente. Los archivos generados son carveables.",
   328→            "refined": "El generador produce archivos válidos. El problema está en el carving, no en el generador.",
   329→        },
   330→        {
   331→            "id": "H-CARVING-005",
   332→            "statement": "Los footers siguen siendo insuficientes",
   333→            "probability": "HIGH",
   334→            "evidence": "PDF: footer %%EOF no incluye \\n. JPEG: footer FF D9 funciona pero deduplicación elimina archivos.",
   335→            "refined": "El footer de PDF es incorrecto (falta \\n). El footer de JPEG funciona pero la deduplicación interfiere.",
   336→        },
   337→        {
   338→            "id": "H-CARVING-002",
   339→            "statement": "El dataset no contiene suficientes formatos carveables",
   340→            "probability": "LOW",
   341→            "evidence": "Todos los formatos probados son carveables. El problema es la extracción, no la detección.",
   342→        },
   343→        {
   344→            "id": "H-CARVING-004",
   345→            "statement": "El Judge penaliza excesivamente el carving",
   346→            "probability": "LOW",
   347→            "evidence": "Cuando el carving extrae datos correctos (ZIP/DOCX/PNG), el Judge los puntúa correctamente.",
   348→        },
   349→        {
   350→            "id": "H-CARVING-007",
   351→            "statement": "La implementación del carving todavía está incompleta",
   352→            "probability": "MEDIUM",
   353→            "evidence": "El scanner funciona (encuentra firmas), pero la extracción tiene bugs (PDF footer, JPEG dedup).",
   354→            "refined": "El carving está parcialmente implementado: la detección funciona, la extracción tiene bugs.",
   355→        },
   356→        {
   357→            "id": "H-CARVING-006",
   358→            "statement": "El RVS/FQS favorecen tipos de archivo que carving no puede recuperar",
   359→            "probability": "LOW",
   360→            "evidence": "Cuando carving funciona (ZIP/DOCX/PNG), RVS/FQS son correctos. No hay sesgo en el scoring.",
   361→        },
   362→    ]
   363→
   364→    prob_order = {"HIGH": 0, "MEDIUM": 1, "UNDETERMINED": 2, "LOW": 3}
   365→    hypothesis_ranking.sort(key=lambda h: prob_order.get(h["probability"], 99))
   366→    diagnosis["hypothesis_ranking"] = hypothesis_ranking
   367→
   368→    return diagnosis

   422→        lines.append(f"- **Causa**: {rc['cause']}")
   423→        lines.append(f"- **Impacto**: {rc['impact']}")
   424→        lines.append(f"- **Severidad**: {rc['severity']}")
   425→        lines.append(f"- **Corrección sugerida**: {rc['fix']}")
   426→        lines.append(f"")
   427→
   428→    # ── Per-format analysis ──
   429→    lines.append(f"## 4. Análisis por Formato")
   430→    lines.append(f"")
   431→
   432→    for fmt in FORMAT_DATASETS:
   433→        fmt_id = fmt["id"]
   434→        if fmt_id not in diagnosis["format_analysis"]:
   435→            continue
   436→        fa = diagnosis["format_analysis"][fmt_id]
   437→        lines.append(f"### Dataset {fmt_id}: {fmt['name']}")
   438→        lines.append(f"")

   446→        lines.append(f"")
   447→
   448→        if fa["per_file_details"]:
   449→            lines.append(f"**Detalle por archivo carved:**")

   453→                fix = " [TRUNCATION FIXABLE]" if d["truncation_fixes_sha256"] else ""
