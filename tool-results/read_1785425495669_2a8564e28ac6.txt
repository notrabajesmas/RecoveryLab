   299→   299→class MotorCarving(BaseMotor):
   300→   300→    """
   301→   301→    Motor Carving: signature-based file recovery.
   302→   302→
   303→   303→    This motor NEVER reads the MFT. It scans the raw disk image
   304→   304→    byte-by-byte looking for known file signatures.
   305→   305→
   306→   306→    Strategy:
   307→   307→      1. Scan the ENTIRE image for file signatures
   308→   308→      2. When a signature is found, extract the file
   309→   309→      3. Use footer markers to determine file boundaries
   310→   310→      4. If no footer, use max-size heuristic
   311→   311→      5. Assign generic names (carved_0001.jpg, etc.)
   312→   312→
   313→   313→    This is the "adversarial baseline" — it represents the best
   314→   314→    you can do without any filesystem metadata.
   315→   315→    """
   316→   316→
   317→   317→    @property
   318→   318→    def name(self) -> str:
   319→   319→        return "Motor Carving (Signature-Only)"
   320→   320→
   321→   321→    @property
   322→   322→    def description(self) -> str:
   323→   323→        return ("Carving: scans raw bytes for file signatures. "
   324→   324→                "NEVER reads MFT, VBR, or any filesystem metadata. "
   325→   325→                "No filenames, no directories, no fragmentation support.")
   326→   326→
   327→   327→    def recover(self, image: bytes, manifest: Dict,
   328→   328→                read_budget: int = 0,
   329→   329→                corruption_metadata: Optional[Dict] = None) -> MotorResult:
   330→   330→        """
   331→   331→        Run signature-based carving recovery.
   332→   332→
   333→   333→        Key constraint: This method NEVER accesses:
   334→   334→          - manifest["mft"] (MFT data)
   335→   335→          - manifest["vbr"] (VBR data)
   336→   336→          - Any filesystem metadata
   337→   337→
   338→   338→        It ONLY uses manifest["cluster_size"] and manifest["total_clusters"]
   339→   339→        to understand the image geometry, and manifest["files"] to know
   340→   340→        what file types exist (for reporting purposes only — NOT for recovery).
   341→   341→
   342→   342→        Actually, to be STRICTLY pure carving, we should NOT use the manifest
   343→   343→        at all for recovery. The manifest is only used for Judge evaluation
   344→   344→        after recovery.
   345→   345→        """
   346→   346→        result = MotorResult(motor_name=self.name)
   347→   347→        cluster_size = manifest.get("cluster_size", 4096)
   348→   348→        total_clusters = manifest.get("total_clusters", len(image) // cluster_size)
   349→   349→
   350→   350→        reads = 0
   351→   351→        sectors_per_cluster = cluster_size // 512
   352→   352→        first_file_reads = 0
   353→   353→        found_first_file = False
   354→   354→
   355→   355→        # ─── Phase 1: Scan entire image for signatures ────────────────
   356→   356→        # We read every cluster sequentially (like a real carver would)
   357→   357→        # This is the MAXIMUM read cost — no metadata guidance at all
   358→   358→
   359→   359→        carved_files: List[Dict] = []
   360→   360→        signatures_found: Dict[str, int] = {}
   361→   361→        scan_offset = 0
   362→   362→
   363→   363→        # Build a list of all header patterns to search for
   364→   364→        # We scan byte-by-byte through the image
   365→   365→        image_len = len(image)
   366→   366→
   367→   367→        # Track which clusters we've already read (for read counting)
   368→   368→        read_clusters: Set[int] = set()
   369→   369→
   370→   370→        # Scan the image looking for file signatures
   371→   371→        # We scan at every byte position within the first sector of each cluster
   372→   372→        # (in practice, signatures are usually at the start of a cluster)
   373→   373→        for cluster_num in range(total_clusters):
   374→   374→            cluster_start = cluster_num * cluster_size
   375→   375→
   376→   376→            if cluster_start + cluster_size > image_len:
   377→   377→                break
   378→   378→
   379→   379→            # Read this cluster
   380→   380→            cluster_data = image[cluster_start:cluster_start + cluster_size]
   381→   381→            read_clusters.add(cluster_num)
   382→   382→            reads += sectors_per_cluster
   383→   383→
   384→   384→            # Check for budget exhaustion
   385→   385→            if read_budget > 0 and reads >= read_budget:
   386→   386→                break
   387→   387→
   388→   388→            # Search for signatures within this cluster
   389→   389→            # Check at the start of the cluster (most common case)
   390→   390→            # and also at every offset within the first 512 bytes
   391→   391→            # (some signatures might be offset within a cluster)
   392→   392→            for sig in SIGNATURES:
   393→   393→                # Quick check: does the signature match at the start?
   394→   394→                matches = self._find_signature_matches(
   395→   395→                    image, cluster_start, cluster_size, sig, image_len
   396→   396→                )
   397→   397→
   398→   398→                for match_offset in matches:
   399→   399→                    # Found a signature! Try to extract the file
   400→   400→                    carved = self._carve_file(
   401→   401→                        image, match_offset, sig, image_len,
   402→   402→                        cluster_size, total_clusters, read_clusters
   403→   403→                    )
   404→   404→
   405→   405→                    if carved is not None:
   406→   406→                        carved_files.append(carved)
   407→   407→                        sig_name = sig.name
   408→   408→                        signatures_found[sig_name] = signatures_found.get(sig_name, 0) + 1
   409→   409→
   410→   410→                        # Update read count for any additional clusters we read
   411→   411→                        # beyond the current scan position
   412→   412→                        extra_reads = carved.get("extra_clusters_read", 0)
   413→   413→                        reads += extra_reads * sectors_per_cluster
   414→   414→
   415→   415→        # ─── Phase 2: Resolve ZIP/DOCX ambiguity ─────────────────────
   416→   416→        # ZIP and DOCX share the same PK\x03\x04 header.
   417→   417→        # We try to distinguish them by looking for internal structure.
   418→   418→        carved_files = self._resolve_zip_docx(carved_files)
   419→   419→
   420→   420→        # ─── Phase 3: Deduplicate overlapping carves ──────────────────
   421→   421→        # If two signatures overlap (e.g., ZIP and DOCX at same offset),
   422→   422→        # keep the more specific one
   423→   423→        carved_files = self._deduplicate_carves(carved_files)
   424→   424→
   425→   425→        # ─── Phase 4: Build result ────────────────────────────────────
   426→   426→        # Assign generic names
   427→   427→        type_counters: Dict[str, int] = {}
   428→   428→
   429→   429→        for carved in carved_files:
   430→   430→            ext = carved["extension"]
   431→   431→            type_counters[ext] = type_counters.get(ext, 0) + 1
   432→   432→            name = f"carved_{type_counters[ext]:04d}{ext}"
   433→   433→
   434→   434→            data = carved["data"]
   435→   435→            sha256 = hashlib.sha256(data).hexdigest()
   436→   436→
   437→   437→            if not found_first_file:
   438→   438→                first_file_reads = reads
   439→   439→                found_first_file = True
   440→   440→
   441→   441→            result.recovered_files.append(RecoveredFile(
   442→   442→                name=name,
   443→   443→                sha256=sha256,
   444→   444→                size=len(data),
   445→   445→                data=data,
   446→   446→                source="carving",  # Mark as carving-recovered
   447→   447→                read_count=reads,
   448→   448→            ))
   449→   449→
   450→   450→        # ─── Compute final metrics ────────────────────────────────────
   451→   451→        result.read_count = reads
   452→   452→        result.time_to_first_file = first_file_reads
   453→   453→        result.mft_entries_parsed = 0  # NEVER parses MFT
   454→   454→        result.total_time_seconds = reads * 0.001
   455→   455→        result.directories_rebuilt = 0  # No directory support
   456→   456→
   457→   457→        # Sectors wasted: we read everything, but not all clusters had files
   458→   458→        # In carving, "wasted" is relative — we had to read everything to find files
   459→   459→        useful_clusters = len(set(
   460→   460→            cf.get("start_cluster", 0) // cluster_size
   461→   461→            for cf in carved_files
   462→   462→        ))
   463→   463→        result.sectors_wasted = max(0, reads - useful_clusters * sectors_per_cluster)
   464→   464→
   465→   465→        # Store carving statistics for debugging
   466→   466→        result.carving_stats = {
   467→   467→            "signatures_found": signatures_found,
   468→   468→            "files_carved": len(carved_files),
   469→   469→            "total_clusters_scanned": len(read_clusters),
   470→   470→            "scan_coverage_pct": len(read_clusters) / total_clusters if total_clusters else 0,
   471→   471→        }
   472→   472→
   473→   473→        return result

   475→   475→    def _find_signature_matches(self, image: bytes, cluster_start: int,
   476→   476→                                 cluster_size: int, sig: FileSignature,
   477→   477→                                 image_len: int) -> List[int]:
   478→   478→        """
   479→   479→        Find all occurrences of a signature within a cluster.
   480→   480→
   481→   481→        Returns list of absolute offsets where the signature was found.
   482→   482→        """
   483→   483→        matches = []
   484→   484→
   485→   485→        # Check at the start of the cluster (most common)
   486→   486→        # and at a few key offsets within the first 512 bytes
   487→   487→        check_offsets = [0]
   488→   488→
   489→   489→        # Also check at sector-aligned offsets within the cluster
   490→   490→        for sector_offset in range(512, min(cluster_size, 4096), 512):
   491→   491→            check_offsets.append(sector_offset)
   492→   492→
   493→   493→        for offset_in_cluster in check_offsets:
   494→   494→            abs_offset = cluster_start + offset_in_cluster
   495→   495→            if abs_offset + sig.header_len > image_len:
   496→   496→                continue
   497→   497→
   498→   498→            # Check if header matches (with mask)
   499→   499→            candidate = image[abs_offset:abs_offset + sig.header_len]
   500→   500→            if self._matches_with_mask(candidate, sig.header, sig.header_mask):
   501→   501→                matches.append(abs_offset)
   502→   502→
   503→   503→        return matches

   505→   505→    def _matches_with_mask(self, candidate: bytes, header: bytes,
   506→   506→                            mask: bytes) -> bool:
   507→   507→        """Check if candidate bytes match the header pattern with mask."""
   508→   508→        if len(candidate) < len(header):
   509→   509→            return False
   510→   510→        for i in range(len(header)):
   511→   511→            if mask[i] == 0xFF:  # Must match
   512→   512→                if candidate[i] != header[i]:
   513→   513→                    return False
   514→   514→            # 0x00 = don't care, skip
   515→   515→        return True

   517→   517→    def _carve_file(self, image: bytes, start_offset: int,
   518→   518→                     sig: FileSignature, image_len: int,
   519→   519→                     cluster_size: int, total_clusters: int,
   520→   520→                     read_clusters: Set[int]) -> Optional[Dict]:
   521→   521→        """
   522→   522→        Extract a file from the image starting at the given offset.
   523→   523→
   524→   524→        Uses footer markers to determine file boundaries.
   525→   525→        If no footer is found, uses max-size heuristic.
   526→   526→        """
   527→   527→        # Check minimum size
   528→   528→        remaining = image_len - start_offset
   529→   529→        if remaining < sig.min_size:
   530→   530→            return None
   531→   531→
   532→   532→        # Determine file end
   533→   533→        file_data = None
   534→   534→        extra_clusters_read = 0
   535→   535→
   536→   536→        if sig.footer:
   537→   537→            # Search for footer after the header
   538→   538→            # Start searching from after the header
   539→   539→            footer_search_start = start_offset + sig.header_len
   540→   540→            max_search_end = min(start_offset + sig.max_size, image_len)
   541→   541→
   542→   542→            # Search for footer
   543→   543→            footer_offset = self._find_footer(
   544→   544→                image, footer_search_start, max_search_end, sig.footer
   545→   545→            )
   546→   546→
   547→   547→            if footer_offset is not None:
   548→   548→                # Found footer — extract file including footer
   549→   549→                file_end = footer_offset + len(sig.footer)
   550→   550→                file_data = image[start_offset:file_end]
   551→   551→
   552→   552→                # Track additional clusters read beyond the scan
   553→   553→                end_cluster = (file_end - 1) // cluster_size
   554→   554→                start_cluster = start_offset // cluster_size
   555→   555→                for c in range(start_cluster, end_cluster + 1):
   556→   556→                    if c not in read_clusters:
   557→   557→                        read_clusters.add(c)
   558→   558→                        extra_clusters_read += 1
   559→   559→            else:
   560→   560→                # No footer found — use max-size heuristic
   561→   561→                # But cap at a reasonable size
   562→   562→                heuristic_size = min(sig.max_size, remaining)
   563→   563→                file_data = image[start_offset:start_offset + heuristic_size]
   564→   564→        else:
   565→   565→            # No footer defined — use max-size heuristic
   566→   566→            # For MP4, try to parse the ftyp box to get the size
   567→   567→            if sig.name == "MP4":
   568→   568→                file_data = self._carve_mp4(image, start_offset, image_len, sig)
   569→   569→            else:
   570→   570→                heuristic_size = min(sig.max_size, remaining)
   571→   571→                file_data = image[start_offset:start_offset + heuristic_size]
   572→   572→
   573→   573→        if file_data is None or len(file_data) < sig.min_size:
   574→   574→            return None
   575→   575→
   576→   576→        # Validate: check if file is too large (likely a false positive)
   577→   577→        if len(file_data) > sig.max_size:
   578→   578→            file_data = file_data[:sig.max_size]
   579→   579→
   580→   580→        return {
   581→   581→            "data": file_data,
   582→   582→            "start_offset": start_offset,
   583→   583→            "start_cluster": start_offset // cluster_size,
   584→   584→            "extension": sig.extension,
   585→   585→            "format_name": sig.name,
   586→   586→            "extra_clusters_read": extra_clusters_read,
   587→   587→            "size": len(file_data),
   588→   588→        }

   590→   590→    def _find_footer(self, image: bytes, start: int, end: int,
   591→   591→                      footer: bytes) -> Optional[int]:
   592→   592→        """
   593→   593→        Find the first occurrence of footer bytes in the image.
   594→   594→
   595→   595→        Uses a simple scan. For large images, this could be optimized
   596→   596→        with a Boyer-Moore search, but for our 10MB images it's fine.
   597→   597→        """
   598→   598→        footer_len = len(footer)
   599→   599→        if footer_len == 0:
   600→   600→            return None
   601→   601→
   602→   602→        # Search in chunks to avoid excessive memory usage
   603→   603→        chunk_size = 1024 * 1024  # 1MB chunks
   604→   604→        pos = start
   605→   605→
   606→   606→        while pos < end:
   607→   607→            chunk_end = min(pos + chunk_size + footer_len, end + footer_len)
   608→   608→            chunk = image[pos:chunk_end]
   609→   609→
   610→   610→            idx = chunk.find(footer)
   611→   611→            if idx != -1:
   612→   612→                return pos + idx
   613→   613→
   614→   614→            pos += chunk_size
   615→   615→
   616→   616→        return None

   618→   618→    def _carve_mp4(self, image: bytes, start_offset: int,
   619→   619→                    image_len: int, sig: FileSignature) -> Optional[bytes]:
   620→   620→        """
   621→   621→        Carve an MP4 file by parsing the ftyp box to determine size.
   622→   622→
   623→   623→        MP4 structure: a series of boxes (atoms), each with:
   624→   624→          [4 bytes size][4 bytes type][data...]
   625→   625→        The first box is typically 'ftyp'.
   626→   626→        """
   627→   627→        # The 'ftyp' is at offset 4 within the first box
   628→   628→        # So the first box starts at start_offset - 4
   629→   629→        box_start = start_offset - 4
   630→   630→
   631→   631→        if box_start < 0 or box_start + 8 > image_len:
   632→   632→            # Can't parse box header — use max-size heuristic
   633→   633→            remaining = min(sig.max_size, image_len - start_offset)
   634→   634→            return image[start_offset:start_offset + remaining]
   635→   635→
   636→   636→        # Read first box size
   637→   637→        try:
   638→   638→            box_size = struct.unpack('>I', image[box_start:box_start + 4])[0]
   639→   639→        except:
   640→   640→            remaining = min(sig.max_size, image_len - start_offset)
   641→   641→            return image[start_offset:start_offset + remaining]
   642→   642→
   643→   643→        if box_size == 0:
   644→   644→            # Box extends to end of file
   645→   645→            return image[box_start:image_len]
   646→   646→        elif box_size == 1:
   647→   647→            # 64-bit extended size
   648→   648→            if box_start + 16 > image_len:
   649→   649→                remaining = min(sig.max_size, image_len - start_offset)
   650→   650→                return image[start_offset:start_offset + remaining]
   651→   651→            box_size = struct.unpack('>Q', image[box_start + 8:box_start + 16])[0]
   652→   652→
   653→   653→        # Sanity check
   654→   654→        if box_size > sig.max_size or box_size < 8:
   655→   655→            remaining = min(sig.max_size, image_len - start_offset)
   656→   656→            return image[start_offset:start_offset + remaining]
   657→   657→
   658→   658→        end = box_start + box_size
   659→   659→        if end > image_len:
   660→   660→            end = image_len
   661→   661→
   662→   662→        return image[box_start:end]

   664→   664→    def _resolve_zip_docx(self, carved_files: List[Dict]) -> List[Dict]:
   665→   665→        """
   666→   666→        Resolve ZIP/DOCX/XLSX ambiguity.
   667→   667→
   668→   668→        All three share the PK\x03\x04 header. We try to distinguish them
   669→   669→        by looking for internal structure markers:
   670→   670→          - DOCX contains "word/" or "word/document.xml"
   671→   671→          - XLSX contains "xl/" or "xl/workbook.xml"
   672→   672→          - ZIP contains other content
   673→   673→
   674→   674→        If we find multiple carves at the same offset (ZIP/DOCX/XLSX),
   675→   675→        we keep the one that matches best.
   676→   676→        """
   677→   677→        # Group by start offset
   678→   678→        by_offset: Dict[int, List[Dict]] = {}
   679→   679→        for cf in carved_files:
   680→   680→            offset = cf["start_offset"]
   681→   681→            if offset not in by_offset:
   682→   682→                by_offset[offset] = []
   683→   683→            by_offset[offset].append(cf)
   684→   684→
   685→   685→        resolved = []
   686→   686→        for offset, files in by_offset.items():
   687→   687→            if len(files) == 1:
   688→   688→                resolved.append(files[0])
   689→   689→                continue
   690→   690→
   691→   691→            # Multiple carves at same offset — likely ZIP/DOCX/XLSX overlap
   692→   692→            # Check internal content for format-specific markers
   693→   693→            data = files[0]["data"]
   694→   694→
   695→   695→            has_docx_markers = (
   696→   696→                b'word/' in data or
   697→   697→                b'word/document.xml' in data or
   698→   698→                b'Content_Types.xml' in data
   699→   699→            )
   700→   700→
   701→   701→            has_xlsx_markers = (
   702→   702→                b'xl/' in data or
   703→   703→                b'xl/workbook.xml' in data or
   704→   704→                b'xl/worksheets/' in data
   705→   705→            )
   706→   706→
   707→   707→            if has_xlsx_markers:
   708→   708→                # Keep the XLSX version
   709→   709→                for f in files:
   710→   710→                    if f["format_name"] == "XLSX":
   711→   711→                        resolved.append(f)
   712→   712→                        break
   713→   713→                else:
   714→   714→                    # No XLSX entry found — convert the first ZIP to XLSX
   715→   715→                    files[0]["extension"] = ".xlsx"
   716→   716→                    files[0]["format_name"] = "XLSX"
   717→   717→                    resolved.append(files[0])
   718→   718→            elif has_docx_markers:
   719→   719→                # Keep the DOCX version
   720→   720→                for f in files:
   721→   721→                    if f["format_name"] == "DOCX":
   722→   722→                        resolved.append(f)
   723→   723→                        break
   724→   724→                else:
   725→   725→                    # No DOCX entry found — convert the ZIP to DOCX
   726→   726→                    files[0]["extension"] = ".docx"
   727→   727→                    files[0]["format_name"] = "DOCX"
   728→   728→                    resolved.append(files[0])
   729→   729→            else:
   730→   730→                # Keep the ZIP version
   731→   731→                for f in files:
   732→   732→                    if f["format_name"] == "ZIP":
   733→   733→                        resolved.append(f)
   734→   734→                        break
   735→   735→                else:
   736→   736→                    resolved.append(files[0])
   737→   737→
   738→   738→        return resolved

   740→   740→    def _deduplicate_carves(self, carved_files: List[Dict]) -> List[Dict]:
   741→   741→        """
   742→   742→        Remove duplicate carves that overlap significantly.
   743→   743→
   744→   744→        If two carves overlap by >50% of the smaller one, keep the larger one.
   745→   745→        """
   746→   746→        if len(carved_files) <= 1:
   747→   747→            return carved_files
   748→   748→
   749→   749→        # Sort by start offset
   750→   750→        sorted_files = sorted(carved_files, key=lambda f: f["start_offset"])
   751→   751→
   752→   752→        deduped = []
   753→   753→        for cf in sorted_files:
   754→   754→            is_duplicate = False
   755→   755→
   756→   756→            for existing in deduped:
   757→   757→                # Check overlap
   758→   758→                overlap_start = max(cf["start_offset"], existing["start_offset"])
   759→   759→                overlap_end = min(
   760→   760→                    cf["start_offset"] + cf["size"],
   761→   761→                    existing["start_offset"] + existing["size"]
   762→   762→                )
   763→   763→
   764→   764→                if overlap_start < overlap_end:
   765→   765→                    overlap_size = overlap_end - overlap_start
   766→   766→                    smaller_size = min(cf["size"], existing["size"])
   767→   767→
   768→   768→                    if overlap_size > smaller_size * 0.5:
   769→   769→                        # Significant overlap — keep the larger one
   770→   770→                        if cf["size"] > existing["size"]:
   771→   771→                            deduped.remove(existing)
   772→   772→                            deduped.append(cf)
   773→   773→                        is_duplicate = True
   774→   774→                        break
   775→   775→
   776→   776→            if not is_duplicate:
   777→   777→                deduped.append(cf)
   778→   778→
   779→   779→        return deduped
