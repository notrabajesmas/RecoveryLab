     3→"""
     4→RecoveryLab — Simulated NTFS Benchmark Engine
     5→================================================
     6→A pure-Python simulation that models the core property we want to test:
     7→Does reading MFT-first recover more files than sequential reading
     8→when a disk is failing?
     9→
    10→The simulation models:
    11→- NTFS disk with MFT entries mapping files to clusters
    12→- A "disk health" model where sectors fail over time
    13→- Two reading strategies: Sequential (Motor A) vs MFT-first (Motor B)
    14→- Measurement: how many files can be recovered before the disk dies
    15→
    16→This is NOT a real NTFS parser. It's a scientific instrument for
    17→testing a specific hypothesis.
    18→"""
    19→
    20→import os
    21→import json
    22→import hashlib
    23→import random
    24→import time
    25→import struct
    26→from dataclasses import dataclass, field, asdict
    27→from typing import List, Dict, Tuple, Optional
    28→from pathlib import Path
    29→
    30→# ─── Constants ────────────────────────────────────────────────────────────────
    31→SECTOR_SIZE = 512          # bytes per sector
    32→CLUSTER_SIZE = 4096        # bytes per cluster (8 sectors)
    33→SECTORS_PER_CLUSTER = CLUSTER_SIZE // SECTOR_SIZE
    34→MFT_ENTRY_SIZE = 1024      # bytes per MFT entry
    35→MFT_ENTRIES_PER_CLUSTER = CLUSTER_SIZE // MFT_ENTRY_SIZE  # 4
    36→
    37→LAB_DIR = Path(__file__).parent
    38→
    39→# ─── Data Models ──────────────────────────────────────────────────────────────
    40→
    41→@dataclass
    42→class SimFile:
    43→    """A simulated file on the disk."""
    44→    file_id: int
    45→    name: str
    46→    size: int                    # bytes
    47→    start_cluster: int
    48→    cluster_count: int
    49→    clusters: List[int]          # list of cluster numbers
    50→    is_resident: bool = False    # if True, data is in MFT entry itself
    51→    checksum: str = ""           # SHA-256 of content
    52→    content: bytes = b""         # simulated content
    53→
    54→@dataclass
    55→class SimMFTEntry:
    56→    """A simulated MFT entry."""
    57→    file_id: int
    58→    is_directory: bool
    59→    is_deleted: bool
    60→    is_in_use: bool
    61→    cluster_start: int           # where this MFT entry lives on disk (cluster)
    62→    file_ref: Optional[SimFile]  # the file this entry describes
    63→
    64→@dataclass
    65→class SimDisk:
    66→    """A simulated NTFS disk."""
    67→    total_clusters: int
    68→    cluster_size: int = CLUSTER_SIZE
    69→    mft_start_cluster: int = 0
    70→    mft_entry_count: int = 0
    71→    mft_entries: List[SimMFTEntry] = field(default_factory=list)
    72→    files: List[SimFile] = field(default_factory=list)
    73→    # Sector health: 1.0 = healthy, 0.0 = dead
    74→    sector_health: List[float] = field(default_factory=list)
    75→    # Which sectors contain MFT data
    76→    mft_sectors: set = field(default_factory=set)
    77→    # Which sectors contain file data
    78→    data_sectors: Dict[int, set] = field(default_factory=dict)  # file_id -> set of sectors
    79→
    80→    @property
    81→    def total_sectors(self):
    82→        return self.total_clusters * SECTORS_PER_CLUSTER
    83→
    84→    @property
    85→    def total_bytes(self):
    86→        return self.total_clusters * self.cluster_size
    87→
    88→    @property
    89→    def mft_cluster_count(self):
    90→        return (self.mft_entry_count * MFT_ENTRY_SIZE + self.cluster_size - 1) // self.cluster_size

    95→def generate_test_files(count: int, min_size: int = 100, max_size: int = 50000) -> List[Dict]:
    96→    """Generate metadata for test files with realistic distribution."""
    97→    file_types = [
    98→        ("photo_{:04d}.jpg", 0.3, 2000, 8000),      # 30% photos, 2-8KB
    99→        ("document_{:04d}.pdf", 0.25, 1000, 20000),   # 25% docs, 1-20KB
   100→        ("video_{:04d}.mp4", 0.1, 50000, 200000),     # 10% videos, 50-200KB (simulated small)
   101→        ("email_{:04d}.eml", 0.15, 500, 5000),        # 15% emails, 0.5-5KB
   102→        ("spreadsheet_{:04d}.xlsx", 0.1, 2000, 30000),# 10% spreadsheets
   103→        ("code_{:04d}.py", 0.1, 200, 10000),          # 10% code files
   104→    ]
   105→
   106→    files = []
   107→    idx = 0
   108→    for pattern, prob, fmin, fmax in file_types:
   109→        n = int(count * prob)
   110→        for i in range(n):
   111→            name = pattern.format(idx)
   112→            size = random.randint(fmin, fmax)
   113→            files.append({"name": name, "size": size, "file_id": idx})
   114→            idx += 1
   115→
   116→    # Fill remaining
   117→    while len(files) < count:
   118→        name = f"file_{idx:04d}.bin"
   119→        size = random.randint(min_size, max_size)
   120→        files.append({"name": name, "size": size, "file_id": idx})
   121→        idx += 1
   122→
   123→    return files[:count]

   126→def create_disk(total_mb: int = 100, file_count: int = 200) -> SimDisk:
   127→    """Create a simulated NTFS disk with files and MFT."""
   128→    total_clusters = (total_mb * 1024 * 1024) // CLUSTER_SIZE
   129→    disk = SimDisk(total_clusters=total_clusters)
   130→
   131→    # MFT starts at cluster 2 (typical NTFS)
   132→    disk.mft_start_cluster = 2
   133→
   134→    # Generate test files
   135→    file_metadata = generate_test_files(file_count)
   136→    total_sectors = disk.total_sectors
   137→
   138→    # Initialize sector health (all healthy)
   139→    disk.sector_health = [1.0] * total_sectors
   140→
   141→    # Allocate clusters for files
   142→    # Data starts after MFT area (reserve clusters 2-100 for MFT)
   143→    next_data_cluster = 200
   144→
   145→    for fm in file_metadata:
   146→        clusters_needed = (fm["size"] + CLUSTER_SIZE - 1) // CLUSTER_SIZE
   147→        # Small files are resident (data in MFT entry)
   148→        is_resident = fm["size"] < 800  # ~800 bytes threshold
   149→
   150→        if is_resident:
   151→            clusters = []
   152→        else:
   153→            # Allocate contiguous or fragmented clusters
   154→            if random.random() < 0.7:  # 70% contiguous
   155→                start = next_data_cluster
   156→                clusters = list(range(start, start + clusters_needed))
   157→                next_data_cluster += clusters_needed
   158→            else:  # 30% fragmented
   159→                clusters = []
   160→                for _ in range(clusters_needed):
   161→                    c = next_data_cluster + random.randint(0, 50)
   162→                    clusters.append(c)
   163→                    next_data_cluster = max(next_data_cluster, c + 1)
   164→
   165→        # Generate content
   166→        content = os.urandom(min(fm["size"], 4096))  # We only need enough to verify
   167→        if fm["size"] > 4096:
   168→            content = content + b'\x00' * (fm["size"] - 4096)
   169→            content = content[:fm["size"]]
   170→        checksum = hashlib.sha256(content).hexdigest()
   171→
   172→        sf = SimFile(
   173→            file_id=fm["file_id"],
   174→            name=fm["name"],
   175→            size=fm["size"],
   176→            start_cluster=clusters[0] if clusters else 0,
   177→            cluster_count=len(clusters),
   178→            clusters=clusters,
   179→            is_resident=is_resident,
   180→            checksum=checksum,
   181→            content=content[:64],  # Store first 64 bytes for verification
   182→        )
   183→        disk.files.append(sf)
   184→
   185→        # Map file's data sectors
   186→        file_sectors = set()
   187→        for c in clusters:
   188→            for s in range(SECTORS_PER_CLUSTER):
   189→                sector = c * SECTORS_PER_CLUSTER + s
   190→                if sector < total_sectors:
   191→                    file_sectors.add(sector)
   192→        disk.data_sectors[sf.file_id] = file_sectors
   193→
   194→    # Create MFT entries
   195→    mft_cluster = disk.mft_start_cluster
   196→    mft_sector_offset = 0
   197→
   198→    for i, sf in enumerate(disk.files):
   199→        entry = SimMFTEntry(
   200→            file_id=sf.file_id,
   201→            is_directory=False,
   202→            is_deleted=False,
   203→            is_in_use=True,
   204→            cluster_start=mft_cluster,
   205→            file_ref=sf,
   206→        )
   207→        disk.mft_entries.append(entry)
   208→
   209→        # MFT entry occupies sectors within its cluster
   210→        entry_sector = mft_cluster * SECTORS_PER_CLUSTER + (i % MFT_ENTRIES_PER_CLUSTER)
   211→        for s in range(2):  # Each MFT entry spans ~2 sectors
   212→            sec = entry_sector + s
   213→            if sec < total_sectors:
   214→                disk.mft_sectors.add(sec)
   215→
   216→    disk.mft_entry_count = len(disk.mft_entries)
   217→
   218→    return disk

   221→# ─── Damage Models ─────────────────────────────────────────────────────────────
   222→
   223→def apply_mft_damage(disk: SimDisk, damage_pct: float) -> SimDisk:
   224→    """Damage a percentage of MFT sectors. Returns modified disk."""
   225→    mft_list = sorted(disk.mft_sectors)
   226→    n_damage = int(len(mft_list) * damage_pct)
   227→    damaged = random.sample(mft_list, min(n_damage, len(mft_list)))
   228→    for sec in damaged:
   229→        disk.sector_health[sec] = 0.0
   230→    return disk

   233→def apply_bad_sectors(disk: SimDisk, pct: float, region: str = "random") -> SimDisk:
   234→    """Add bad sectors. region: 'random', 'data_area', 'beginning'."""
   235→    total = disk.total_sectors
   236→    n_bad = int(total * pct)
   237→
   238→    if region == "random":
   239→        bad = random.sample(range(total), n_bad)
   240→    elif region == "data_area":
   241→        # Avoid MFT area (clusters 2-100)
   242→        data_start = 200 * SECTORS_PER_CLUSTER
   243→        bad = random.sample(range(data_start, total), n_bad)
   244→    elif region == "beginning":
   245→        bad = list(range(n_bad))
   246→    else:
   247→        bad = random.sample(range(total), n_bad)
   248→
   249→    for sec in bad:
   250→        disk.sector_health[sec] = 0.0
   251→    return disk

   254→def apply_growing_failure(disk: SimDisk, start_pct: float, rate: float) -> SimDisk:
   255→    """Simulate a disk that's failing progressively.
   256→    Sectors near the beginning fail first (common in real drives).
   257→    start_pct: initial failure percentage
   258→    rate: additional failure per 1000 sectors read
   259→    """
   260→    total = disk.total_sectors
   261→    n_initial = int(total * start_pct)
   262→    # Fail sectors from the end of the disk (common pattern)
   263→    for i in range(n_initial):
   264→        sec = total - 1 - i
   265→        disk.sector_health[sec] = 0.0
   266→    return disk

   269→# ─── Reading Strategies ────────────────────────────────────────────────────────
   270→
   271→@dataclass
   272→class ReadResult:
   273→    """Result of a read attempt."""
   274→    sector: int
   275→    success: bool
   276→    data_recovered: bool
   277→    is_mft: bool = False
   278→    file_id: Optional[int] = None
   279→    read_time_ms: float = 0.0
   280→
   281→
   282→@dataclass
   283→class RecoveryResult:
   284→    """Result of a complete recovery session."""
   285→    strategy: str
   286→    total_sectors_read: int = 0
   287→    successful_reads: int = 0
   288→    failed_reads: int = 0
   289→    mft_entries_recovered: int = 0
   290→    mft_entries_total: int = 0
   291→    files_recovered: int = 0
   292→    files_total: int = 0
   293→    files_recoverable: int = 0  # files whose MFT entry was read
   294→    bytes_recovered: int = 0
   295→    bytes_total: int = 0
   296→    disk_died: bool = False
   297→    sectors_before_death: int = 0
   298→    time_elapsed_ms: float = 0.0
   299→    # For tracking what was recovered
   300→    recovered_file_ids: set = field(default_factory=set)
   301→    recovered_mft_ids: set = field(default_factory=set)
   302→
   303→    @property
   304→    def file_recovery_rate(self) -> float:
   305→        return self.files_recovered / max(self.files_total, 1)
   306→
   307→    @property
   308→    def mft_recovery_rate(self) -> float:
   309→        return self.mft_entries_recovered / max(self.mft_entries_total, 1)
   310→
   311→    @property
   312→    def data_recovery_rate(self) -> float:
   313→        return self.bytes_recovered / max(self.bytes_total, 1)
   314→
   315→    def to_dict(self) -> dict:
   316→        d = asdict(self)
   317→        d['file_recovery_rate'] = self.file_recovery_rate
   318→        d['mft_recovery_rate'] = self.mft_recovery_rate
   319→        d['data_recovery_rate'] = self.data_recovery_rate
   320→        return d

   322→
   323→def read_sector(disk: SimDisk, sector: int, fail_probability_growth: float = 0.0,
   324→                sectors_read_so_far: int = 0) -> Tuple[bool, bool]:
   325→    """Try to read a sector. Returns (success, data_recovered).
   326→    
   327→    fail_probability_growth: if > 0, simulates growing failure as more sectors are read.
   328→    Each read has a small chance of killing the disk.
   329→    """
   330→    if sector >= len(disk.sector_health):
   331→        return False, False
   332→
   333→    # Base health
   334→    health = disk.sector_health[sector]
   335→
   336→    # Growing failure: each read has a small chance of damaging nearby sectors
   337→    if fail_probability_growth > 0:
   338→        # Probability of disk dying increases with reads
   339→        death_prob = fail_probability_growth * (sectors_read_so_far / 1000.0)
   340→        if random.random() < death_prob:
   341→            # Disk dies! Kill a bunch of remaining sectors
   342→            return False, True  # (read failed, disk died)
   343→
   344→    return health > 0.0, False

   347→def motor_a_sequential(disk: SimDisk, max_reads: int = None,
   348→                       fail_growth: float = 0.0) -> RecoveryResult:
   349→    """Motor A: Sequential reading. Read sectors 0, 1, 2, 3...
   350→    This is how ddrescue works by default.
   351→    """
   352→    result = RecoveryResult(strategy="Motor A: Sequential")
   353→    result.files_total = len(disk.files)
   354→    result.mft_entries_total = len(disk.mft_entries)
   355→    result.bytes_total = sum(f.size for f in disk.files)
   356→
   357→    total_sectors = len(disk.sector_health)
   358→    max_reads = max_reads or total_sectors
   359→    disk_died = False
   360→
   361→    # Build sector -> file mapping for data sectors
   362→    sector_to_file = {}
   363→    for fid, sectors in disk.data_sectors.items():
   364→        for sec in sectors:
   365→            sector_to_file[sec] = fid
   366→
   367→    # Build sector -> MFT entry mapping
   368→    sector_to_mft = {}
   369→    for entry in disk.mft_entries:
   370→        entry_sectors = set()
   371→        base = entry.cluster_start * SECTORS_PER_CLUSTER
   372→        for s in range(2):
   373→            sec = base + s
   374→            if sec in disk.mft_sectors:
   375→                entry_sectors.add(sec)
   376→        for sec in entry_sectors:
   377→            sector_to_mft[sec] = entry.file_id
   378→
   379→    # Track which MFT entries are fully read (need all their sectors)
   380→    mft_sectors_needed = {}  # file_id -> set of sectors
   381→    for entry in disk.mft_entries:
   382→        entry_sectors = set()
   383→        base = entry.cluster_start * SECTORS_PER_CLUSTER
   384→        for s in range(2):
   385→            sec = base + s
   386→            if sec in disk.mft_sectors:
   387→                entry_sectors.add(sec)
   388→        if entry_sectors:
   389→            mft_sectors_needed[entry.file_id] = entry_sectors
   390→
   391→    mft_sectors_read = {}  # file_id -> set of sectors read
   392→
   393→    # Track which file sectors have been read
   394→    file_sectors_read = {}  # file_id -> set of sectors read
   395→
   396→    # Read sequentially
   397→    for sector in range(min(total_sectors, max_reads)):
   398→        success, died = read_sector(disk, sector, fail_growth, result.total_sectors_read)
   399→        result.total_sectors_read += 1
   400→
   401→        if died:
   402→            disk_died = True
   403→            break
   404→
   405→        if success:
   406→            result.successful_reads += 1
   407→
   408→            # Check if this is an MFT sector
   409→            if sector in sector_to_mft:
   410→                fid = sector_to_mft[sector]
   411→                if fid not in mft_sectors_read:
   412→                    mft_sectors_read[fid] = set()
   413→                mft_sectors_read[fid].add(sector)
   414→
   415→            # Check if this is a data sector
   416→            if sector in sector_to_file:
   417→                fid = sector_to_file[sector]
   418→                if fid not in file_sectors_read:
   419→                    file_sectors_read[fid] = set()
   420→                file_sectors_read[fid].add(sector)
   421→        else:
   422→            result.failed_reads += 1
   423→
   424→    # Count recovered MFT entries
   425→    for fid, needed in mft_sectors_needed.items():
   426→        if fid in mft_sectors_read and mft_sectors_read[fid] >= needed:
   427→            result.mft_entries_recovered += 1
   428→            result.recovered_mft_ids.add(fid)
   429→
   430→    # Count recovered files
   431→    # A file is "recovered" if:
   432→    # 1. Its MFT entry was read (we know the file exists and where its clusters are)
   433→    # 2. All its data sectors were read
   434→    # For resident files, only MFT entry is needed
   435→    for sf in disk.files:
   436→        if sf.file_id in result.recovered_mft_ids:
   437→            result.files_recoverable += 1
   438→            if sf.is_resident:
   439→                # Resident file: MFT entry contains the data
   440→                result.files_recovered += 1
   441→                result.bytes_recovered += sf.size
   442→                result.recovered_file_ids.add(sf.file_id)
   443→            else:
   444→                # Non-resident: need all data sectors
   445→                needed = disk.data_sectors.get(sf.file_id, set())
   446→                read = file_sectors_read.get(sf.file_id, set())
   447→                if needed and read >= needed:
   448→                    result.files_recovered += 1
   449→                    result.bytes_recovered += sf.size
   450→                    result.recovered_file_ids.add(sf.file_id)
   451→
   452→    result.disk_died = disk_died
   453→    result.sectors_before_death = result.total_sectors_read if disk_died else 0
   454→
   455→    return result

   458→def motor_b_mft_first(disk: SimDisk, max_reads: int = None,
   459→                      fail_growth: float = 0.0) -> RecoveryResult:
   460→    """Motor B: MFT-first reading. Read MFT sectors first, then data.
   461→    After reading MFT, prioritize data sectors for files whose MFT was recovered.
   462→    """
   463→    result = RecoveryResult(strategy="Motor B: MFT-first")
   464→    result.files_total = len(disk.files)
   465→    result.mft_entries_total = len(disk.mft_entries)
   466→    result.bytes_total = sum(f.size for f in disk.files)
   467→
   468→    total_sectors = len(disk.sector_health)
   469→    max_reads = max_reads or total_sectors
   470→    disk_died = False
   471→
   472→    # Build sector -> file mapping
   473→    sector_to_file = {}
   474→    for fid, sectors in disk.data_sectors.items():
   475→        for sec in sectors:
   476→            sector_to_file[sec] = fid
   477→
   478→    # Build MFT entry -> sectors mapping
   479→    mft_sectors_needed = {}
   480→    for entry in disk.mft_entries:
   481→        entry_sectors = set()
   482→        base = entry.cluster_start * SECTORS_PER_CLUSTER
   483→        for s in range(2):
   484→            sec = base + s
   485→            if sec in disk.mft_sectors:
   486→                entry_sectors.add(sec)
   487→        if entry_sectors:
   488→            mft_sectors_needed[entry.file_id] = entry_sectors
   489→
   490→    mft_sectors_read = {}
   491→    file_sectors_read = {}
   492→
   493→    # ── PHASE 1: Read MFT sectors first ──
   494→    mft_sector_list = sorted(disk.mft_sectors)
   495→    for sector in mft_sector_list:
   496→        if result.total_sectors_read >= max_reads:
   497→            break
   498→
   499→        success, died = read_sector(disk, sector, fail_growth, result.total_sectors_read)
   500→        result.total_sectors_read += 1
   501→
   502→        if died:
   503→            disk_died = True
   504→            break
   505→
   506→        if success:
   507→            result.successful_reads += 1
   508→            # Check which MFT entry this belongs to
   509→            for fid, needed in mft_sectors_needed.items():
   510→                if sector in needed:
   511→                    if fid not in mft_sectors_read:
   512→                        mft_sectors_read[fid] = set()
   513→                    mft_sectors_read[fid].add(sector)
   514→        else:
   515→            result.failed_reads += 1
   516→
   517→    # Count recovered MFT entries after Phase 1
   518→    recovered_mft_ids = set()
   519→    for fid, needed in mft_sectors_needed.items():
   520→        if fid in mft_sectors_read and mft_sectors_read[fid] >= needed:
   521→            recovered_mft_ids.add(fid)
   522→
   523→    # ── PHASE 2: Read data sectors for files whose MFT was recovered ──
   524→    # Prioritize data sectors belonging to recovered files
   525→    priority_data_sectors = set()
   526→    for sf in disk.files:
   527→        if sf.file_id in recovered_mft_ids and not sf.is_resident:
   528→            for sec in disk.data_sectors.get(sf.file_id, set()):
   529→                priority_data_sectors.add(sec)
   530→
   531→    # Also read MFT sectors we haven't read yet (in case some were missed)
   532→    # But prioritize data sectors first
   533→    read_sectors = set(mft_sector_list)  # Already read in Phase 1
   534→
   535→    # Sort priority sectors (prefer smaller files first for quick wins)
   536→    priority_list = sorted(priority_data_sectors - read_sectors)
   537→
   538→    for sector in priority_list:
   539→        if result.total_sectors_read >= max_reads:
   540→            break
   541→        if disk_died:
   542→            break
   543→
   544→        success, died = read_sector(disk, sector, fail_growth, result.total_sectors_read)
   545→        result.total_sectors_read += 1
   546→        read_sectors.add(sector)
   547→
   548→        if died:
   549→            disk_died = True
   550→            break
   551→
   552→        if success:
   553→            result.successful_reads += 1
   554→            if sector in sector_to_file:
   555→                fid = sector_to_file[sector]
   556→                if fid not in file_sectors_read:
   557→                    file_sectors_read[fid] = set()
   558→                file_sectors_read[fid].add(sector)
   559→        else:
   560→            result.failed_reads += 1
   561→
   562→    # ── PHASE 3: Read remaining sectors sequentially ──
   563→    for sector in range(total_sectors):
   564→        if result.total_sectors_read >= max_reads:
   565→            break
   566→        if disk_died:
   567→            break
   568→        if sector in read_sectors:
   569→            continue
   570→
   571→        success, died = read_sector(disk, sector, fail_growth, result.total_sectors_read)
   572→        result.total_sectors_read += 1
   573→        read_sectors.add(sector)
   574→
   575→        if died:
   576→            disk_died = True
   577→            break
   578→
   579→        if success:
   580→            result.successful_reads += 1
   581→            if sector in sector_to_file:
   582→                fid = sector_to_file[sector]
   583→                if fid not in file_sectors_read:
   584→                    file_sectors_read[fid] = set()
   585→                file_sectors_read[fid].add(sector)
   586→        else:
   587→            result.failed_reads += 1
   588→
   589→    # ── Count results ──
   590→    result.mft_entries_recovered = len(recovered_mft_ids)
   591→    result.recovered_mft_ids = recovered_mft_ids
   592→
   593→    for sf in disk.files:
   594→        if sf.file_id in recovered_mft_ids:
   595→            result.files_recoverable += 1
   596→            if sf.is_resident:
   597→                result.files_recovered += 1
   598→                result.bytes_recovered += sf.size
   599→                result.recovered_file_ids.add(sf.file_id)
   600→            else:
   601→                needed = disk.data_sectors.get(sf.file_id, set())
   602→                read = file_sectors_read.get(sf.file_id, set())
   603→                if needed and read >= needed:
   604→                    result.files_recovered += 1
   605→                    result.bytes_recovered += sf.size
   606→                    result.recovered_file_ids.add(sf.file_id)
   607→
   608→    result.disk_died = disk_died
   609→    result.sectors_before_death = result.total_sectors_read if disk_died else 0
   610→
   611→    return result
   612→

   614→# ─── Benchmark Runner ──────────────────────────────────────────────────────────

   616→@dataclass
   617→class Scenario:
   618→    """A benchmark scenario."""
   619→    name: str
   620→    description: str
   621→    disk_mb: int = 100
   622→    file_count: int = 200
   623→    damage_type: str = "mft"          # mft, bad_sectors, growing, combined
   624→    damage_params: dict = field(default_factory=dict)
   625→    fail_growth: float = 0.0          # probability of disk dying per 1000 reads
   626→    max_reads: int = 0                # 0 = unlimited
   627→    runs: int = 5                     # number of runs to average
   628→
   629→    def create_disk(self) -> SimDisk:
   630→        disk = create_disk(total_mb=self.disk_mb, file_count=self.file_count)
   631→        if self.damage_type == "mft":
   632→            apply_mft_damage(disk, self.damage_params.get("pct", 0.3))
   633→        elif self.damage_type == "bad_sectors":
   634→            apply_bad_sectors(disk, self.damage_params.get("pct", 0.01),
   635→                            self.damage_params.get("region", "random"))
   636→        elif self.damage_type == "growing":
   637→            apply_growing_failure(disk, self.damage_params.get("start_pct", 0.0),
   638→                                self.damage_params.get("rate", 0.0))
   639→        elif self.damage_type == "combined":
   640→            apply_mft_damage(disk, self.damage_params.get("mft_pct", 0.2))
   641→            apply_bad_sectors(disk, self.damage_params.get("bad_pct", 0.005))
   642→        elif self.damage_type == "healthy":
   643→            pass  # No damage
   644→        return disk
   645→
   646→
   647→def run_scenario(scenario: Scenario) -> List[Dict]:
   648→    """Run a scenario with both motors and return results."""
   649→    results = []
   650→
   651→    for run in range(scenario.runs):
   652→        random.seed(42 + run)  # Reproducible but different per run
   653→
   654→        disk = scenario.create_disk()
   655→
   656→        # Create a copy of disk for Motor B (since reading modifies state)
   657→        import copy
   658→        disk_b = copy.deepcopy(disk)
   659→
   660→        # Run Motor A
   661→        max_reads = scenario.max_reads or disk.total_sectors
   662→        result_a = motor_a_sequential(disk, max_reads=max_reads,
   663→                                      fail_growth=scenario.fail_growth)
   664→
   665→        # Run Motor B
   666→        result_b = motor_b_mft_first(disk_b, max_reads=max_reads,
   667→                                     fail_growth=scenario.fail_growth)
   668→
   669→        results.append({
   670→            "run": run,
   671→            "scenario": scenario.name,
   672→            "motor_a": result_a.to_dict(),
   673→            "motor_b": result_b.to_dict(),
   674→        })
   675→
   676→    return results
