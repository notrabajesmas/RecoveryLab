   151→def compute_codebase_hash() -> str:
   152→    """Compute a hash of all Python source files in the RecoveryLab."""
   153→    hasher = hashlib.sha256()
   154→    py_files = sorted(RECOVERYLAB_ROOT.rglob("*.py"))
   155→
   156→    for py_file in py_files:
   157→        # Skip __pycache__ and experiment output
   158→        if "__pycache__" in str(py_file) or "output" in str(py_file):
   159→            continue
   160→        try:
   161→            with open(py_file, 'rb') as f:
   162→                hasher.update(f.read())
   163→        except Exception:
   164→            pass
   165→
   166→    return hasher.hexdigest()[:16]

   169→def compute_result_hash(metrics_dict: Dict) -> str:
   170→    """Compute a deterministic hash of key metrics."""
   171→    hash_fields = {
   172→        "files_recovered": metrics_dict.get("files_recovered"),
   173→        "files_correct_checksum": metrics_dict.get("files_correct_checksum"),
   174→        "read_count": metrics_dict.get("read_count"),
   175→        "recovery_rate": metrics_dict.get("recovery_rate"),
   176→        "rvs": round(metrics_dict.get("rvs", 0), 6),
   177→        "weighted_functional_score": round(metrics_dict.get("weighted_functional_score", 0), 6),
   178→    }
   179→    hash_str = json.dumps(hash_fields, sort_keys=True)
   180→    return hashlib.sha256(hash_str.encode()).hexdigest()[:16]

   183→def run_single_execution(run_index: int, image: bytes, manifest: Dict,
   184→                         motor_name: str = "MFT-First") -> Dict:
   185→    """Run a single execution and return all measurements."""
   186→    if motor_name == "MFT-First":
   187→        motor = MotorBMFTFirst()
   188→    elif motor_name == "Carving":
   189→        motor = MotorCarving()
   190→    else:
   191→        raise ValueError(f"Unknown motor: {motor_name}")
   192→
   193→    judge = RecoveryJudge(manifest)
   194→
   195→    t_start = time.perf_counter()
   196→    result = motor.recover(image, manifest, read_budget=0)
   197→    t_end = time.perf_counter()
   198→    runtime_ms = (t_end - t_start) * 1000.0
   199→
   200→    judge_input = [{
   201→        "name": f.name,
   202→        "sha256": f.sha256,
   203→        "size": f.size,
   204→        "is_directory": f.is_directory,
   205→        "data": f.data,
   206→    } for f in result.recovered_files]
   207→
   208→    metrics = judge.judge(
   209→        recovered_files=judge_input,
   210→        read_count=result.read_count,
   211→        sectors_wasted=result.sectors_wasted,
   212→        time_to_first_file=result.time_to_first_file,
   213→        mft_entries_parsed=result.mft_entries_parsed,

   216→    utility = compute_overall_utility(metrics.rvs, metrics.weighted_functional_score)
   217→    metrics_dict = metrics.to_dict()
   218→
   219→    return {
   220→        "run": run_index,
   221→        "motor": motor_name,
   222→        "timestamp": datetime.datetime.now().isoformat(),
   223→        "overall_utility": utility["overall_utility"],
   224→        "rvs": metrics.rvs,

   227→        "read_count": metrics.read_count,

   230→        "files_recovered": metrics_dict.get("files_recovered", 0),
   231→        "files_correct_checksum": metrics_dict.get("files_correct_checksum", 0),
